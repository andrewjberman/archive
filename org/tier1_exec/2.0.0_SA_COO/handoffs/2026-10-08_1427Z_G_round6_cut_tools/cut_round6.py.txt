def add_judgment(path, mech_ids, jid, cell, text):
    t = open(path, encoding="utf-8").read()
    old = "<expected_criteria>%s</expected_criteria>" % ",".join(mech_ids)
    assert t.count(old) == 1, old
    allids = ",".join(mech_ids + [jid])
    t = t.replace(old, '<criterion id="%s" cell_refs="%s" judgment="true"/>\n`%s` (judgment) -- %s You record the artifacts\' paths and NO verdict.\n\n<expected_criteria>%s</expected_criteria>' % (jid, cell, jid, text, allids))
    t = t.replace("Criterion ids: %s." % ", ".join(mech_ids), "Criterion ids: %s, %s (judgment: recorded UNVERIFIABLE-BY-INSTRUMENT with a status_reason naming `3.0.0_SA_CVO`)." % (", ".join(mech_ids), jid))
    t = t.replace("<instance> %s`" % ",".join(mech_ids), "<instance> %s`" % allids)
    t = t.replace("- **Refusal clause.**", "- The judgment criterion gets its artifact paths and NO verdict from you.\n- **Refusal clause.**", 1)
    assert t.count(allids) >= 2
    open(path, "w", encoding="utf-8", newline="\n").write(t)
