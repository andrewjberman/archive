def tags(s):
    i, j = s.find('<proof_of_spec>'), s.find('</proof_of_spec>')
    b = s[i:j]
    return (len(re.findall(r'<criterion id="AC\d+"', b)), b.count('</criterion>'),
            b.count('<invocation>'), b.count('</invocation>'), b.count('<expect>'), b.count('</expect>'))
