data "aws_iam_policy_document" "irpt_directory_refresh" {
  statement {
    sid       = "ReadFirmDirectoryPrefixOnly"
    effect    = "Allow"
    actions   = ["s3:GetObject"]
    resources = local.directory_object_arns
  }

  # PutObject ONLY, and ONLY under _refresh/. No delete and no overwrite of a directory artifact.
  statement {
    sid       = "WriteRefreshManifestsOnly"
    effect    = "Allow"
    actions   = ["s3:PutObject"]
    resources = local.refresh_manifest_arns
  }

  # ---------------------------------------------------------------------------------------
  # THE ABSENCE ORACLE (cell IB-rt-6, row IRPT_BCKEND-W2-S50 -- measured in production, not
  # reasoned about). This role had NO ListBucket by design: "the manifest key is computed from
  # the business date, never discovered". That design is UNCHANGED and this grant does not
  # weaken it -- the calendar still chooses the date and the code still computes the key.
  #
  # WHAT IT FIXES: S3 answers a GetObject on a MISSING key with 403 AccessDenied, NOT 404, when
  # the caller cannot list the bucket. So the first real fire of this function raised on its own
  # first read for all three firms and wrote nothing: a genuine absence arrived wearing a
  # denial's face, and the classifier refused to call it absence -- correctly, because it cannot
  # tell them apart. Two instruments agreed on the cause: the live invoke returned
  # manifest_read_error=AccessDenied for 3 of 3 firms, and simulate-principal-policy (single
  # ARN, exact grain) reported s3:GetObject on that exact manifest key ALLOWED while
  # s3:ListBucket on the bucket was implicitDeny.
  #
  # WHY THIS IS NOT A WIDENING. The condition binds the list to EXACTLY the prefix the GetObject
  # statement above already grants -- the function may now list precisely the set it may already
  # read, and not one key more. Both values are DERIVED from `local.directory_prefix`, so a
  # prefix change moves the read grant and the list grant together instead of leaving one behind.
  # ---------------------------------------------------------------------------------------
  statement {
    sid       = "ListExactlyWhatItMayAlreadyRead"
    effect    = "Allow"
    actions   = ["s3:ListBucket"]
    resources = local.firm_bucket_arns

    condition {
      test     = "StringLike"
      variable = "s3:prefix"
      values = [
        "${local.directory_prefix}/*",
        "${local.directory_prefix}/",
      ]
    }
  }

  statement {
    sid       = "OwnLogStreamOnly"
    effect    = "Allow"
    actions   = ["logs:CreateLogStream", "logs:PutLogEvents"]
    resources = ["arn:aws:logs:us-east-1:${data.aws_caller_identity.current.account_id}:log-group:/aws/lambda/irpt-directory-refresh:*"]
  }

  # =========================================================================================
  # CELL IB-rt-6 -- THE BUILD DISPATCH. FOUR STATEMENTS, AND THE SPLIT IS NOT COSMETIC:
  # ssm:SendCommand and ec2:StartInstances DO support resource-level permissions and are
  # pinned to the ONE instance (and the ONE AWS-owned document) this cell may reach;
  # ssm:GetCommandInvocation, ssm:DescribeInstanceInformation and ec2:DescribeInstances DO NOT
  # support resource-level permissions at all, so their "*" is the IAM plane's own limit and is
  # declared as such rather than left looking like a careless wildcard. Separating them makes
  # the difference READABLE in the policy document instead of only in a reviewer's memory.
  #
  # WHAT IS DELIBERATELY ABSENT: no ssm:CancelCommand, no ssm:SendCommand on any other instance,
  # no ec2:StopInstances and no ec2:TerminateInstances. This function starts a host it finds
  # stopped and NEVER stops one -- the mdpf daily chain owns that lifecycle and a refresh that
  # shut the box down could kill a chain run mid-flight (the accrual bridge's own header says
  # exactly this). No iam:PassRole: the command runs under the instance's own profile.
  # =========================================================================================
  statement {
    sid     = "DispatchTheDirectoryBuildToOneRunnerOnly"
    effect  = "Allow"
    actions = ["ssm:SendCommand"]
    resources = [
      local.irpt_runner_instance_arn,
      "arn:aws:ssm:us-east-1::document/AWS-RunShellScript",
    ]
  }

  statement {
    sid     = "StartThatRunnerNeverStopIt"
    effect  = "Allow"
    actions = ["ec2:StartInstances"]
    # ONE instance. A refresh that could start an arbitrary instance is a different blast radius
    # from one that can wake the runner it already dispatches to.
    resources = [local.irpt_runner_instance_arn]
  }

  statement {
    sid    = "ReadBackTheCommandsThisFunctionSent"
    effect = "Allow"
    # THE PERISHABILITY GRANT. Without this the cell could dispatch and never learn the outcome,
    # and a REFRESH_DISPATCHED manifest would sit at RUNNING forever. These three actions carry
    # NO resource-level support in IAM (AWS service reference: "resource types: none"), so the
    # wildcard is the plane's, not this policy's laxity, and all three are READ-ONLY.
    actions = [
      "ssm:GetCommandInvocation",
      "ssm:DescribeInstanceInformation",
      "ec2:DescribeInstances",
    ]
    resources = ["*"]
  }
}

# B7 part 2 (option B) -- THREE CONCURRENT KEEP-WARM FIRES for irpt-directory-search per 5-minute
# tick (4.0.0_SA_CRO, 2026-09-24; CTO 100230 "OPTION B GO ... an EventBridge/scheduler rule with the
# same function as three concurrent targets")
#
# NAME: irpt-directory-keepwarm-5min. The SCHEDULE grammar ends in a cadence token; "5min" is the
# registry row added to org/standards/NAMING_SCHEMA.json cadence.values for this schedule (CTO ruling
# IRPT-LISTING-SPEED-PIN-GO-20260924), never a hand-typed workaround.
# =============================================================================
#
# Speed brief e4b28c17c4 s.2.3 lever B7, approved by Andrew 2026-09-23 ("I like all your recs, yes go
# ahead", @6fc3f5f785); CTO assignment IRPT-LISTING-SPEED-ASSIGNMENT-20260924 item A: "B7 keep-warm on
# the IRPT routes (a scheduled ping is not a monitor; Andrew approved always-on in the same word)".
# Measured cause (brief s.2.2 row 4): 18% of irpt-directory-search calls were cold starts at 2.3-6.7 s
# each, multiplied by the page's fan-out of three parallel browses. Measured after the single ping
# (Andrew's HAR 04:32Z): the first browse of a fan-out was warm and the second and third were not,
# because ONE ping keeps ONE container. Hence three.
#
# WHY A RULE WITH THREE TARGETS AND NOT THREE SCHEDULES: three `rate(5 minutes)` Scheduler schedules
# each start their clock at creation, so they fire minutes apart and each fire lands on the one
# container the previous fire just freed. An EventBridge rule dispatches ALL its targets at the
# same match, so the three fires are concurrent -- and each fire HOLDS its container for
# `irpt_warm_hold_ms` (served.py warm branch, capped at 2000) so a 2 ms answer cannot let Lambda
# reuse one container for all three. The hold is carried in the target input, so raising it is a
# plan on this file, never a code change.
#
# WHAT A FIRE DOES: invokes irpt-directory-search DIRECTLY with {"irpt_warm": true, ...}. served.py's
# lambda_entry answers {"warm": true, "function": ...} BEFORE the boundary, the authorizer chain or
# any S3 read, so the fire keeps one initialised container resident and touches no data. The marker
# cannot be reached through the HTTP API (API Gateway builds the event's top level itself), so a
# browser caller cannot produce a warm event.
#
# THE RECEIPT: one `{"event": "irpt_warm", "slot": n, ...}` log line per fire, so a tick reads as
# three lines in THREE DISTINCT log streams (a log stream is a container). Fewer distinct streams
# than three on a tick = the hold is too short for that tick's dispatch jitter; raise the input.
#
# NOT A MONITOR, NOT AN ALARM: nothing reads the answer, nothing is notified, no metric is graded.
# Cost: rate(5 minutes) x 3 = 864 fires/day of ~0.5 s each at 1769 MB, about 0.03 USD/day.
# ENABLED, and the standing bound is satisfied rather than waived: the fire reaches one Lambda whose
# warm branch reads nothing and whose import closure holds no vendor client.
#
# REPLACES the single Scheduler schedule + its invoke-only role from the first cut of this block
# (5886d0b9bd, applied 04:21Z): that shape sustains one container, and the rule below sustains three
# on its own, so keeping both would only put a fourth, differently-timed fire into every tick.

resource "aws_cloudwatch_event_rule" "irpt_directory_keepwarm" {
  name                = "irpt-directory-keepwarm-5min"
  description         = "IRPT B7 option B: three concurrent keep-warm fires for irpt-directory-search every 5 minutes, one per target. The function answers before the boundary and reads nothing; not a monitor, nothing is notified."
  schedule_expression = "rate(5 minutes)"
  state               = "ENABLED"
}

resource "aws_cloudwatch_event_target" "irpt_directory_keepwarm" {
  count = 3

  rule      = aws_cloudwatch_event_rule.irpt_directory_keepwarm.name
  target_id = "irpt-directory-keepwarm-slot-${count.index + 1}"
  arn       = module.irpt_directory_search.function_arn

  input = jsonencode({
    irpt_warm         = true
    irpt_warm_slot    = count.index + 1
    irpt_warm_hold_ms = 500
  })
}

# The rule's permission to invoke the function: EventBridge rules invoke Lambda by resource policy,
# not by an assumed role, and the source_arn pins it to THIS rule -- no other rule, bus or account
# gains an invoke on irpt-directory-search from this statement.
resource "aws_lambda_permission" "irpt_directory_keepwarm_events" {
  statement_id  = "AllowIrptDirectoryKeepwarmRule"
  action        = "lambda:InvokeFunction"
  function_name = module.irpt_directory_search.function_name
  principal     = "events.amazonaws.com"
  source_arn    = aws_cloudwatch_event_rule.irpt_directory_keepwarm.arn
}

output "directory_keepwarm_rule_state" {
  value       = aws_cloudwatch_event_rule.irpt_directory_keepwarm.state
  description = "Must read ENABLED; three targets per fire keep three irpt-directory-search containers resident."
}

output "directory_keepwarm_target_count" {
  value       = length(aws_cloudwatch_event_target.irpt_directory_keepwarm)
  description = "Must read 3 -- one warm fire per target per tick."
}
