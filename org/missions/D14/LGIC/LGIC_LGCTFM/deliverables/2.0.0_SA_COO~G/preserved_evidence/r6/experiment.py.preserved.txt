import json

DENOMINATOR_S3KEY = 154
N_TIER = 23
N_FIRM = 1


def ingest(nodes_raw):
    """Verbatim copy of site (A)."""
    nodes, by_arn, by_tn = {}, {}, {}
    for r in nodes_raw:
        nid = r.get("id")
        if not nid:
            continue
        nodes[nid] = {"node_type": r.get("t"), "name": r.get("name") or nid, "drift": r.get("d") or "unset",
                      "arn_or_path": r.get("arn") or nid, "src": "comp-kuzu"}
        if r.get("arn"):
            by_arn[r["arn"]] = nid
        by_tn.setdefault((r.get("t"), r.get("name")), []).append(nid)
    return nodes, by_arn, by_tn


def make_resolve(nodes, by_arn, by_tn):
    """Verbatim copy of site (B), closing over the same nodes/by_arn/by_tn ingest() built."""
    def resolve(cf_node):
        nid = by_arn.get(cf_node.get("arn_or_path"))
        if nid is None:
            cands = by_tn.get((cf_node["node_type"], cf_node["name"])) or []
            nid = cands[0] if len(cands) == 1 else None
        if nid is None:
            nid = "comp-find:%s:%s" % (cf_node["node_type"], cf_node.get("arn_or_path") or cf_node["name"])
            nodes[nid] = dict(cf_node)
        return nid
    return resolve


def has_bucket(arn_or_path):
    """A resolved value carries a bucket iff it names a real opal-data-* writer bucket."""
    return isinstance(arn_or_path, str) and "opal-data-" in arn_or_path


def cf_node_for(key):
    """Site (C): comp-find's own S3Key band prints no detail line -> arn_or_path falls back to bare
    name, REGARDLESS of what any candidate's arn value is in kuzu."""
    return {"node_type": "S3Key", "name": key, "arn_or_path": key, "drift": "unset", "src": "comp-find"}


def tier_nodes_raw(key, portal_arn=""):
    return [
        {"id": "arn:aws:s3:::opal-data-bd/%s" % key, "arn": "arn:aws:s3:::opal-data-bd/%s" % key,
         "name": key, "t": "S3Key", "d": "unset"},
        {"id": "arn:aws:s3:::opal-data-inf/%s" % key, "arn": "arn:aws:s3:::opal-data-inf/%s" % key,
         "name": key, "t": "S3Key", "d": "unset"},
        {"id": "s3:%s" % key, "arn": portal_arn, "name": key, "t": "S3Key", "d": "unset"},
    ]


def firm_nodes_raw(key, portal_arn=""):
    return [{"id": "s3:%s" % key, "arn": portal_arn, "name": key, "t": "S3Key", "d": "unset"}]


def run(tag, portal_arn_tier, portal_arn_firm):
    tier_bucketless = 0
    for i in range(N_TIER):
        key = "TIER_KEY_%02d" % i
        nodes, by_arn, by_tn = ingest(tier_nodes_raw(key, portal_arn_tier))
        resolve = make_resolve(nodes, by_arn, by_tn)
        nid = resolve(cf_node_for(key))
        if not has_bucket(nodes[nid]["arn_or_path"]):
            tier_bucketless += 1

    firm_bucketless = 0
    key = "blackdiamond/49f/_ops/pull_state.json"
    nodes, by_arn, by_tn = ingest(firm_nodes_raw(key, portal_arn_firm))
    # round 5: this row is read straight off the kuzu-native `nodes` dict, no cf_node/resolve() call.
    nid = "s3:%s" % key
    if not has_bucket(nodes[nid]["arn_or_path"]):
        firm_bucketless += 1

    print("%s: tier_prefixed bucketless=%d/%d  firm_prefixed bucketless=%d/%d" %
          (tag, tier_bucketless, N_TIER, firm_bucketless, N_FIRM))
    return tier_bucketless, firm_bucketless
