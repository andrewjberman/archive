def consent_scope_names(email: str) -> list:
    """The scope names the REAL /login consent leg puts in its authorization URL for `email` (token_broker.build_consent_url over a real state): parsed from the URL's
    `scope` parameter in-process (the URL is never printed), minus the three OIDC scopes MSAL adds itself."""
    url = token_broker.build_consent_url(token_broker.make_state(email))
    q = urllib.parse.parse_qs(urllib.parse.urlparse(url).query)
    return sorted(set(" ".join(q.get("scope", [])).split()) - MSAL_ADDED)

def _spy(self, scopes, account, **kw):
    res = _orig(self, scopes, account, **kw)

def get(path: str):
    """One read-only Graph GET. Returns (HTTP status, parsed JSON or None). The token is only ever placed in the Authorization header."""
    req = urllib.request.Request("https://graph.microsoft.com/v1.0" + path, headers={"Authorization": "Bearer " + token}, method="GET")
    try:
        with urllib.request.urlopen(req, timeout=30) as r:
            raw = r.read()
            return r.status, (json.loads(raw) if raw else None)
    except urllib.error.HTTPError as e:
        return e.code, None
    except Exception:  # noqa: BLE001
        return "ERR", None
