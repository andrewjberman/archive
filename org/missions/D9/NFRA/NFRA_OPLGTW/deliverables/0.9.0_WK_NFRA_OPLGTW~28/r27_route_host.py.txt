"""OG-rl-13 (AMENDMENT A3) form step 7a / verify route arms: the RIV-rt-1 chart route GET /rivalfunds/<day>/chart/<FUND>, and its two neighbours, through the GuardMiddleware stack of the
code in OG_GW_DIR (default the INSTALLED /opt/opal-gateway) on the HOST in a SEPARATE process (never the running service, never /opt/opal-gateway/.env: python-dotenv's load_dotenv is
made a no-op BEFORE app.config imports and the process runs under env -i).
Identity: a SYNTHETIC signed-in address, ES256-signed with a LOCAL test key that replaces the ELB public key fetch ("installed bytes, test signing key"); the email-deny read is stubbed
(no state store). The S3 reads are REAL (the instance role, read-only: GetObject of the published rivalfunds prefix); the audit sink is a SCRATCH jsonl and AUDIT_S3_BUCKET is asserted empty.
Nothing is written to any grant, state store or the lake. Prints codes, byte counts and 16-hex body digests only.

usage: python run.py <expect>     expect = before (the installed r26 code: the chart route exists) | after (the release r27: it still does, and the neighbour pages answer as before)
`after` takes the BEFORE values of the two neighbour pages from env R27_BEFORE_ROUTE (a json object with day_status, day_bytes, day_sha16, audit_status, audit_bytes, audit_sha16) and
sets day_page_unchanged / audit_page_unchanged by comparing status, byte count and body digest to them. Without that env the two flags are None (a rehearsal printing values only).
Prints one line per arm, then R27_ROUTE {...} and ROUTE_ARMS <expect>: n of m  RESULT: PASS|FAIL."""
import hashlib
import json
import os
import sys

expect = sys.argv[1]
gw = os.environ.get("OG_GW_DIR") or "/opt/opal-gateway"
sys.path.insert(0, gw)
scratch = os.environ.get("R27_SCRATCH") or "/tmp"
audit_path = os.path.join(scratch, "r27_route_audit.jsonl")
os.environ["MOCK_GRAPH"] = "1"
os.environ["IDENTITY_PROVIDER"] = "alb_oidc"
os.environ["RATE_LIMIT_ENABLED"] = "0"
os.environ["AUDIT_PATH"] = audit_path
os.environ["AWS_DEFAULT_REGION"] = "us-east-1"
os.environ.pop("AUDIT_S3_BUCKET", None)
os.environ["ENABLED_INTEGRATIONS"] = "ycharts opal_portal globgrph opal_discover portal_index schwab blackdiamond"
PERSON = "synthetic.person@opalinvest.test"
DAY = "2026-10-06"          # a day whose page and audit page exist (0 chart objects exist under the page prefix)
DAY_NONE = "2000-01-01"     # a day with no page objects at all
BEFORE = json.loads(os.environ.get("R27_BEFORE_ROUTE") or "null")

import dotenv  # noqa: E402
dotenv.load_dotenv = lambda *a, **k: False

import jwt                                                       # noqa: E402
from cryptography.hazmat.primitives import serialization        # noqa: E402
from cryptography.hazmat.primitives.asymmetric import ec        # noqa: E402
from starlette.testclient import TestClient                      # noqa: E402

from app import audit, identity, rivalfunds_view, server          # noqa: E402

assert not audit._cfg.AUDIT_S3_BUCKET, "AUDIT_S3_BUCKET must be empty"
print("INSTALLED_FILES server.py md5 %s rivalfunds_view.py md5 %s from %s" % (
    hashlib.md5(open(server.__file__, "rb").read()).hexdigest()[:8], hashlib.md5(open(rivalfunds_view.__file__, "rb").read()).hexdigest()[:8],
    "stage" if os.environ.get("OG_GW_DIR") else "installed"))
key = ec.generate_private_key(ec.SECP256R1())
pub_pem = key.public_key().public_bytes(serialization.Encoding.PEM, serialization.PublicFormat.SubjectPublicKeyInfo).decode()
priv_pem = key.private_bytes(serialization.Encoding.PEM, serialization.PrivateFormat.PKCS8, serialization.NoEncryption())
identity._fetch_elb_public_key = lambda kid: pub_pem


async def _deny(email):
    return None


server._email_denied = _deny


def signed(email):
    return jwt.encode({"email": email, "sub": "local"}, priv_pem, algorithm="ES256", headers={"kid": "local-kid"})


client = TestClient(server.app, raise_server_exceptions=False)
H = {"x-amzn-oidc-data": signed(PERSON)}
route_present = any(getattr(r, "path", None) == "/rivalfunds/{day}/chart/{fund}" for r in server.app.routes)
c404 = client.get("/rivalfunds/%s/chart/ABC" % DAY_NONE, headers=H, follow_redirects=False)
c400 = client.get("/rivalfunds/%s/chart/abc" % DAY, headers=H, follow_redirects=False)
cnoid = client.get("/rivalfunds/%s/chart/ABC" % DAY, follow_redirects=False)
dpage = client.get("/rivalfunds/%s" % DAY, headers=H, follow_redirects=False)
apage = client.get("/rivalfunds/%s/audit" % DAY, headers=H, follow_redirects=False)


def sha16(b):
    return hashlib.sha256(b).hexdigest()[:16]


out = {"mode": expect, "day": DAY, "day_none": DAY_NONE, "chart_route_registered": route_present,
       "chart_404_status": c404.status_code, "chart_404_body_named": b"no chart for this fund and day" in c404.content,
       "chart_400_status": c400.status_code, "chart_no_identity_status": cnoid.status_code,
       "day_status": dpage.status_code, "day_bytes": len(dpage.content), "day_sha16": sha16(dpage.content),
       "audit_status": apage.status_code, "audit_bytes": len(apage.content), "audit_sha16": sha16(apage.content),
       "synthetic_principal": True, "grants_written": 0, "state_store_written": False}
out["chart_404"] = out["chart_404_status"] == 404 and out["chart_404_body_named"]
out["chart_400"] = out["chart_400_status"] == 400
if BEFORE:
    out["day_page_unchanged"] = (out["day_status"], out["day_bytes"], out["day_sha16"]) == (BEFORE["day_status"], BEFORE["day_bytes"], BEFORE["day_sha16"])
    out["audit_page_unchanged"] = (out["audit_status"], out["audit_bytes"], out["audit_sha16"]) == (BEFORE["audit_status"], BEFORE["audit_bytes"], BEFORE["audit_sha16"])
else:
    out["day_page_unchanged"] = None
    out["audit_page_unchanged"] = None
rows = []


def chk(label, ok, detail):
    rows.append(bool(ok))
    print("%s %s: %s" % ("PASS" if ok else "FAIL", label, detail))


if expect == "after":
    chk("route /rivalfunds/{day}/chart/{fund} registered", route_present, "registered %s" % route_present)
    chk("chart_404: a day with no chart -> 404 with 'no chart for this fund and day'", out["chart_404"], "%s, named body %s" % (out["chart_404_status"], out["chart_404_body_named"]))
    chk("chart_400: a lowercase fund -> 400, no S3 read", out["chart_400"], "%s" % out["chart_400_status"])
    chk("no identity -> 401 on the chart route", out["chart_no_identity_status"] == 401, "%s" % out["chart_no_identity_status"])
    if BEFORE:
        chk("day_page_unchanged: /rivalfunds/<day> answers as on the installed code (status, bytes, digest)", out["day_page_unchanged"],
            "%s %s bytes sha16 %s vs before %s %s %s" % (out["day_status"], out["day_bytes"], out["day_sha16"], BEFORE["day_status"], BEFORE["day_bytes"], BEFORE["day_sha16"]))
        chk("audit_page_unchanged: /rivalfunds/<day>/audit answers as on the installed code (status, bytes, digest)", out["audit_page_unchanged"],
            "%s %s bytes sha16 %s vs before %s %s %s" % (out["audit_status"], out["audit_bytes"], out["audit_sha16"], BEFORE["audit_status"], BEFORE["audit_bytes"], BEFORE["audit_sha16"]))
    else:
        chk("day page answers 200 (no BEFORE values supplied)", out["day_status"] == 200, "%s %s bytes" % (out["day_status"], out["day_bytes"]))
        chk("audit page answers 200 (no BEFORE values supplied)", out["audit_status"] == 200, "%s %s bytes" % (out["audit_status"], out["audit_bytes"]))
else:
    chk("route /rivalfunds/{day}/chart/{fund} registered on the installed r26 code", route_present, "registered %s" % route_present)
    chk("chart_404 on the installed code: a day with no chart -> 404 with 'no chart for this fund and day'", out["chart_404"], "%s, named body %s" % (out["chart_404_status"], out["chart_404_body_named"]))
    chk("chart_400 on the installed code: a lowercase fund -> 400", out["chart_400"], "%s" % out["chart_400_status"])
    chk("day page answers 200", out["day_status"] == 200, "%s %s bytes" % (out["day_status"], out["day_bytes"]))
    chk("audit page answers 200", out["audit_status"] == 200, "%s %s bytes" % (out["audit_status"], out["audit_bytes"]))
print("R27_ROUTE " + json.dumps(out, sort_keys=True))
print("ROUTE_ARMS %s: %d of %d  RESULT: %s" % (expect, sum(rows), len(rows), "PASS" if all(rows) else "FAIL"))
sys.exit(0 if all(rows) else 1)
