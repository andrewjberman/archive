class MemStore:
    """HybridStore: POLICY# reads go to the real table (strong, via app.access_state); all else in memory."""

    def __init__(self):
        self.items = {}
        self.put_runtime_calls = 0
        self.policy_reads = 0
        self._real = None

    def get(self, key):
        if str(key.pk).startswith("POLICY#"):
            from app import access_state as acs
            if self._real is None:
                self._real = acs.store()
            self.policy_reads += 1
            return self._real.get(key)
        item = self.items.get((key.pk, key.sk))
        return json.loads(json.dumps(item, default=str)) if item is not None else None

    def put_runtime(self, key, item, condition, names=None, values=None):
        from app import access_state as acs
        if str(key.pk).startswith("POLICY#"):
            raise RuntimeError("GA1 probe refuses a POLICY write")
        self.put_runtime_calls += 1
        if (key.pk, key.sk) in self.items:
            raise acs.ConditionFailed("exists")
        self.items[(key.pk, key.sk)] = json.loads(json.dumps(item, default=str))


class AuditSink:
    def __init__(self):
        self.puts = 0

    def put_object(self, Bucket, Key, Body, ContentType):
        self.puts += 1
        return {"ResponseMetadata": {"HTTPStatusCode": 200, "RequestId": "rl6"}, "ETag": '"rl6"'}


def build(asset_policy=None, counterfactual=False):
    """Returns (service, catalog, store, contract). asset_policy=None, counterfactual=False is the SERVED construction.
    counterfactual=True parses an IN-MEMORY copy of the contract with the two dim_transaction data prefixes added (the file on
    the host is untouched): it answers what the catalog would serve if the contract carried them, and is never the served state."""
    from app import account_ref as aref
    from app.integrations.schwab import aws_roles
    from app.integrations.schwab import catalog as cat
    from app.integrations.schwab import integration as integ
    from app.integrations.schwab.engine import ExecutionEngine
    from app.integrations.schwab import export as _export

    contract = cat.load_contract()
    if counterfactual:
        raw = json.loads(json.dumps(contract.raw))
        raw["data_prefixes"] = list(raw["data_prefixes"]) + ["schwab/%s/silver/dim_transaction/" % m for m in sorted(contract.masters)]
        contract = cat.parse_contract(raw, "<memory: counterfactual>")
    configs = aws_roles.role_configs(contract.raw, aws_roles.environ_with_external_ids(os.environ))
    data_role = aws_roles.AssumedRole(configs[aws_roles.READER])
    catalog_role = aws_roles.AssumedRole(configs[aws_roles.CATALOG])
    store = MemStore()
    catalog = cat.Catalog(contract=contract, catalog_role=catalog_role, data_role=data_role, store=store)
    os.makedirs(os.path.join(SCRATCH, "src"), exist_ok=True)
    fetch = integ.S3SourceFetcher(data_role, contract, temp_root=os.path.join(SCRATCH, "src"))
    engine = ExecutionEngine(temp_root=os.path.join(SCRATCH, "engine"))
    keyring = aref.Keyring("v1", b"\x07" * 32)
    service = integ.SchwabService(
        contract_raw=contract.raw, catalog=catalog, engine=engine, fetch_sources=fetch, store=store,
        audit_client=AuditSink(), audit_bucket="opal-mcp-audit", keyring_provider=lambda: keyring,
        ticket_issuer=_export.make_ticket_issuer(store), asset_policy=asset_policy,
        reference_key=b"\x05" * 32)
    return service, catalog, store, contract


def blank_count(column):
    """Null or all-whitespace string cells in a pyarrow ChunkedArray."""
    import pyarrow as pa
    import pyarrow.compute as pc
    n = column.null_count
    if pa.types.is_string(column.type) or pa.types.is_large_string(column.type):
        trimmed = pc.utf8_trim_whitespace(column)
        n += pc.sum(pc.fill_null(pc.equal(trimmed, ""), False).cast(pa.int64())).as_py() or 0
    return int(n)
