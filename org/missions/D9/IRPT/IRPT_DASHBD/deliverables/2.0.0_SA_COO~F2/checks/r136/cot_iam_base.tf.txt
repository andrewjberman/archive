data "aws_iam_policy_document" "irpt_account_return_daily_execution" {
  statement {
    sid       = "ListSilverTablePrefixesOnly"
    effect    = "Allow"
    actions   = ["s3:ListBucket"]
    resources = ["arn:aws:s3:::${local.irpt_arc_silver_bucket}"]
    condition {
      test     = "StringLike"
      variable = "s3:prefix"
      values = [
        "${local.irpt_arc_silver_allocations}*",
        "${local.irpt_arc_silver_transactions}*",
      ]
    }
  }

  statement {
    sid     = "ReadTwoBlackDiamondSilverTablesOnly"
    effect  = "Allow"
    actions = ["s3:GetObject"]
    resources = [
      "arn:aws:s3:::${local.irpt_arc_silver_bucket}/${local.irpt_arc_silver_allocations}*",
      "arn:aws:s3:::${local.irpt_arc_silver_bucket}/${local.irpt_arc_silver_transactions}*",
    ]
  }

  # PutObject only, on the ONE gold prefix this landing writes (data.parquet and manifest.json of each landed day).
  statement {
    sid       = "WriteTheOneAccountReturnDailyGoldPrefixOnly"
    effect    = "Allow"
    actions   = ["s3:PutObject"]
    resources = ["arn:aws:s3:::${local.irpt_arc_gold_bucket}/irpt/${local.irpt_arc_firm}/gold/account_return_daily/*"]
  }

  # ListBucket on the gold bucket under ONE prefix: the handler's scheduled path lists the account_return_daily prefix to
  # find the newest landed partition (handler.py lines 119 to 120). Amendment 1 of the order (issuer, 2026-10-15).
  statement {
    sid       = "ListTheOneAccountReturnDailyGoldPrefixOnly"
    effect    = "Allow"
    actions   = ["s3:ListBucket"]
    resources = ["arn:aws:s3:::${local.irpt_arc_gold_bucket}"]
    condition {
      test     = "StringLike"
      variable = "s3:prefix"
      values   = ["irpt/${local.irpt_arc_firm}/gold/account_return_daily/*"]
    }
  }

  statement {
    sid       = "OwnLogStreamOnly"
    effect    = "Allow"
    actions   = ["logs:CreateLogStream", "logs:PutLogEvents"]
    resources = ["arn:aws:logs:us-east-1:${data.aws_caller_identity.current.account_id}:log-group:/aws/lambda/${module.irpt_account_return_daily_lambda_name.name}:*"]
  }
}
