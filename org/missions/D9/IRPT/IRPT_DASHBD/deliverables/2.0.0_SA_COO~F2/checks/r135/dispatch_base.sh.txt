#!/usr/bin/env bash
# =========================================================================
# orion-gold-dispatch.sh -- Orion Gold Layer Launcher
# =========================================================================
#
# AGENT QUICK REFERENCE (read this, then run one command):
#
#   Full chain (most common):
#     bash /tmp/orion-gold-dispatch.sh --chain full --bg
#
#   Incremental D-chain only (assumes A/B/C outputs exist):
#     bash /tmp/orion-gold-dispatch.sh --chain incremental --bg
#
#   Resume from a specific script:
#     bash /tmp/orion-gold-dispatch.sh --from INF_D_00 --bg
#
#   Re-run one script:
#     bash /tmp/orion-gold-dispatch.sh --single INF_D_00c
#
#   Dry run (compute, don't write):
#     bash /tmp/orion-gold-dispatch.sh --chain full --dry-run
#
#   Check status / tail logs:
#     bash /tmp/orion-gold-dispatch.sh --status
#
#   Kill everything:
#     bash /tmp/orion-gold-dispatch.sh --kill
#
#   SSM one-liner (copy-paste to EC2):
#     aws s3 cp s3://opal-data-inf/code/gold/orion-gold-dispatch.sh /tmp/orion-gold-dispatch.sh \
#       && bash /tmp/orion-gold-dispatch.sh --chain full --bg
#
# Full flag/mode/chain reference: run `bash orion-gold-dispatch.sh --help`.
# Flag docs live in `show_help` only -- two-source truth drifts.
#
# CHAINS:
#   full        API pulls + silver builds + A_* + B_* + C_* + D_* + INF_09 (all 40 scripts, sequential)
#   incremental API pulls + silver builds + D-chain only
#               Assumes A/B/C gold outputs already exist in S3.
#
# EXECUTION MODEL:
#   Sequential only -- no parallelism. Read-during-write race condition risk
#   eliminated. Each script must complete before the next starts.
#
# WATCHDOG:
#   5-min grace-window pattern (11478af7). Handled by systemd timer
#   (watchdog-shutdown.sh on the EC2 instance). NOT GOLD_KEEP_ALIVE.
#   Do NOT modify this script to add keep-alive markers.
#
# SCRIPT-LOAD CONTRACT (EAGER — do NOT assume lazy per-step):
#   sync_scripts() is called ONCE at run_chain() entry (line ~281). It does a
#   single bulk boto3 download of ALL code/gold/ + code/silver/ + code/bronze/
#   scripts to /tmp/orion_gold_scripts/ before any script executes.
#   Post-boot S3 uploads to code/gold/ will NOT propagate to a running chain.
#   Deploy scripts to S3 BEFORE invoking inf-gold-trigger, not during.
#   --single and --from modes also call sync_scripts() at their entry points.
#
# =========================================================================
set -e

# ---------------------------------------------------------------------------
# Environment
# ---------------------------------------------------------------------------
INF_S3_BUCKET="${INF_S3_BUCKET:-opal-data-inf}"
SCRIPTS="/tmp/orion_gold_scripts"
# GLDRNS.B1 forensics hotfix (2026-07-16, CTO GO 000444Z): /tmp is tmpfs on the runner -- a chain
# that dies before its completion-upload leaves NO log after the watchdog's power cycle (tonight's
# +13s death was unreadable for exactly this reason; 07-11/07-12 likely the same). Logs now live on
# persistent EBS, and the EXIT trap ships the newest log to S3 unconditionally (crash or complete).
LOGDIR="/var/log/orion_gold_logs"
RUN_DATE=$(date -u +%Y-%m-%d)

mkdir -p "$SCRIPTS" "$LOGDIR"

# ---------------------------------------------------------------------------
# Telemetry bootstrap (opal-telemetry.sh + telemetry-sync.sh from opal-runner-shared)
# ADHC_TELEM_WIRE 2026-05-13: structured telemetry library + 30s S3 push loop.
# Pushes live runner state to s3://opal-data-inf/telemetry/inf-gold-layer-runner/status.json
# (Andrew canonical real-time status path -- replaces SSM/CloudWatch poll).
# Soft-fail: chain proceeds without telemetry if bootstrap fails (no-op stubs).
# ---------------------------------------------------------------------------
TELEM_AVAILABLE=0
TELEM_SYNC_PID=""
TELEM_SERVICE="${TELEM_SERVICE:-inf-gold-layer-runner}"

# /var/run requires root; runner runs as root per `sudo shutdown` at end.
# Pre-create world-writable as belt-and-suspenders for non-root invocations.
mkdir -p /var/run/opal-telemetry 2>/dev/null \
    || sudo mkdir -p /var/run/opal-telemetry 2>/dev/null \
    || true
chmod 1777 /var/run/opal-telemetry 2>/dev/null \
    || sudo chmod 1777 /var/run/opal-telemetry 2>/dev/null \
    || true

if aws s3 cp s3://opal-runner-shared/code/shared/opal-telemetry.sh "$SCRIPTS/opal-telemetry.sh" --quiet 2>/dev/null \
   && aws s3 cp s3://opal-runner-shared/code/shared/telemetry-sync.sh "$SCRIPTS/telemetry-sync.sh" --quiet 2>/dev/null; then
    chmod +x "$SCRIPTS/telemetry-sync.sh"
    # shellcheck disable=SC1091
    if source "$SCRIPTS/opal-telemetry.sh" 2>/dev/null; then
        TELEM_AVAILABLE=1
        echo "Telemetry: opal-telemetry.sh sourced from opal-runner-shared (status JSON will push every 30s)"
    fi
fi

if [ "$TELEM_AVAILABLE" != "1" ]; then
    echo "WARN: opal-telemetry.sh bootstrap failed -- status JSON will NOT push to S3 (chain proceeds)"
    # no-op stubs so subsequent calls don't error
    telem_init()       { :; }
    telem_step_begin() { :; }
    telem_step_done()  { :; }
    telem_update()     { :; }
    telem_warn()       { :; }
    telem_error()      { :; }
    telem_complete()   { :; }
fi

# Trap-driven cleanup: kill background sync loop on dispatcher exit + do
# one final telemetry-sync.sh push so terminal-state JSON archives correctly.
_telem_cleanup() {
    if [ -n "$TELEM_SYNC_PID" ]; then
        kill "$TELEM_SYNC_PID" 2>/dev/null || true
    fi
    if [ "$TELEM_AVAILABLE" = "1" ]; then
        bash "$SCRIPTS/telemetry-sync.sh" >/dev/null 2>&1 || true
    fi
}

# ---------------------------------------------------------------------------
# BRONZE-FIX #7: FATAL loud channel (2026-06-12 P0 follow-up f).
# A fail-closed guard (sync_scripts collision FATAL) that dies SILENTLY
# defeats itself: 13:00Z->20:37Z the chain was dead, the dispatch lock was
# held by a dead process, and nothing blared until a human noticed stale
# ADVI. This trap makes any pre-completion non-zero exit LOUD:
#   1. CW metric Inflection/GoldChain.ChainFailed=1 -- this is the EXISTING
#      live convention: alarm inf-gold-chain-failed (alarms.tf) already
#      watches exactly {namespace=Inflection/GoldChain, metric=ChainFailed,
#      dims Service=inf-gold-layer-runner} -> opal-infrastructure-alerts,
#      and iam.tf scopes PutMetricData to THIS namespace. The alarm + IAM
#      landed earlier but the script-side emitter was never added (the
#      half-deploy class again) -- THIS is the missing emitter. A clean
#      complete emits ChainFailed=0 (mdpf finish() parity).
#   2. Dispatch-lock self-release -- a dead chain must not strand the lock
#      (the trigger's stale-TTL clear took 8.1h; this makes it immediate).
#   3. SNS direct publish to opal-critical-surface-alerts -- best-effort
#      belt-and-suspenders (no sns:Publish grant today; the alarm in (1) is
#      the live loud path; topic disposition = AE/CTO per TO).
# Alert egress runs BEFORE log upload / telemetry cleanup -- the 13:00Z
# FATAL died pre-log-upload, so anything sequenced after uploads inherits
# the exact blindness this fixes. Correlation token in the SNS body:
# lock:<run_id> (joins to inf-gold-trigger logs + deploy_ledger --proof,
# format pinned in .tools/deploy_ledger.py 307f7d44e).
# ---------------------------------------------------------------------------
ALERT_SNS_TOPIC="${ALERT_SNS_TOPIC:-arn:aws:sns:us-east-1:313476888478:opal-critical-surface-alerts}"
ALERT_CW_NAMESPACE="Inflection/GoldChain"
ALERT_CW_SERVICE="inf-gold-layer-runner"
ALERT_PIPELINE="inf"
CHAIN_STARTED=0

_alert_instance_id() {
    local tok
    tok=$(curl -s -X PUT "http://169.254.169.254/latest/api/token" \
        -H "X-aws-ec2-metadata-token-ttl-seconds: 60" 2>/dev/null || true)
    curl -s -H "X-aws-ec2-metadata-token: $tok" \
        "http://169.254.169.254/latest/meta-data/instance-id" 2>/dev/null \
        || echo "unknown-instance"
}

fatal_alert() {
    local rc="$1"
    echo "FATAL-ALERT: dispatch exiting rc=$rc before completion -- firing loud channel (BRONZE-FIX #7)"
    # 1) CW metric first: fires the EXISTING inf-gold-chain-failed alarm
    #    (dims must be EXACTLY Service=inf-gold-layer-runner to match it).
    aws cloudwatch put-metric-data --namespace "$ALERT_CW_NAMESPACE" \
        --metric-name ChainFailed --value 1 --unit Count \
        --dimensions "Service=$ALERT_CW_SERVICE" \
        2>/dev/null || echo "WARN: ChainFailed put-metric-data failed"
    # 2) Capture + self-release the dispatch lock (dead holder must not block retriggers).
    local lock_json lock_run_id
    lock_json=$(aws s3 cp "s3://${INF_S3_BUCKET}/gold/dispatch-lock.json" - 2>/dev/null || true)
    lock_run_id=$(echo "$lock_json" | grep -oE '"run_id": *"[^"]*"' | cut -d'"' -f4)
    if [ -n "$lock_json" ]; then
        aws s3 rm "s3://${INF_S3_BUCKET}/gold/dispatch-lock.json" 2>/dev/null \
            && echo "FATAL-ALERT: dispatch lock self-released (lock:${lock_run_id:-unparsed})" \
            || echo "WARN: dispatch lock self-release failed (lock:${lock_run_id:-unparsed})"
    fi
    # 3) SNS publish (best-effort; CW metric above is the guaranteed floor).
    local log_tail=""
    if [ -n "${CURRENT_LOG_FILE:-}" ] && [ -f "${CURRENT_LOG_FILE:-}" ]; then
        log_tail=$(tail -8 "$CURRENT_LOG_FILE" 2>/dev/null || true)
    fi
    aws sns publish --topic-arn "$ALERT_SNS_TOPIC" \
        --subject "RED: ${ALERT_PIPELINE} gold dispatch FATAL rc=${rc} (${CHAIN_LABEL:-unknown})" \
        --message "RED alert-not-block (BRONZE-FIX #7 FATAL loud channel)
pipeline: ${ALERT_PIPELINE}
chain: ${CHAIN_LABEL:-unknown}
rc: ${rc}
run_date: ${RUN_DATE}
instance: $(_alert_instance_id)
lock:${lock_run_id:-none} (self-released by this trap if present)
log tail:
${log_tail:-<none captured>}" \
        2>/dev/null || echo "WARN: sns publish failed/denied -- ChainFATAL CW metric is the floor signal (sns:Publish grant rides AE wiring)"
}

_on_exit() {
    local rc=$?
    if [ "$CHAIN_STARTED" = "1" ]; then
        if [ "$rc" -ne 0 ]; then
            fatal_alert "$rc"
        else
            # Clean complete: ChainFailed=0 (mdpf finish() parity; keeps the
            # alarm's Maximum>=1 semantics + a daily heartbeat datapoint).
            aws cloudwatch put-metric-data --namespace "$ALERT_CW_NAMESPACE" \
                --metric-name ChainFailed --value 0 --unit Count \
                --dimensions "Service=$ALERT_CW_SERVICE" \
                2>/dev/null || true
        fi
    fi
    _telem_cleanup
}
trap _on_exit EXIT

# *** An invocation path (`--single` / `--from`) HAS NO CHAIN LITERAL, so R4 structurally cannot see it and

check_freshness() {
    local silver_latest gold_as_of
    silver_latest=$(aws s3 ls "s3://${INF_S3_BUCKET}/silver/aum_snapshot/" 2>/dev/null \
        | grep -oE '[0-9]{4}-[0-9]{2}-[0-9]{2}' | sort | tail -1)
    gold_as_of=$(aws s3 cp "s3://${INF_S3_BUCKET}/gold/artifacts/header_metrics.json" - 2>/dev/null \
        | python3 -c 'import json,sys; print(json.load(sys.stdin).get("data_as_of", ""))' 2>/dev/null)

    if [ -z "$silver_latest" ]; then
        echo "  WARN: Could not determine silver latest date -- proceeding."
        return 0
    fi
    if [ -z "$gold_as_of" ]; then
        echo "  No prior gold data_as_of found -- proceeding (first run)."
        return 0
    fi

    if [ "$silver_latest" \> "$gold_as_of" ]; then
        echo "  Data freshness: silver=$silver_latest > gold=$gold_as_of -- proceeding."
        return 0
    fi

    echo "============================================================"
    echo "SKIP: No new silver data since last gold run."
    echo "  Silver latest:   $silver_latest"
    echo "  Gold data_as_of: $gold_as_of"
    echo "  Pass --force to override and run anyway."
    echo "============================================================"
    return 1
}

sync_scripts() {
    echo "Syncing scripts from s3://${INF_S3_BUCKET}/code/gold/ ..."
    # sync_scripts() flattens four S3 code/ prefixes into one working dir
    # keyed by basename. A bare last-write-wins flatten silently runs whichever
    # same-named copy downloaded last -- ADVI_INFETL.SYNC-COLLISION 2026-05-18:
    # a stale code/bronze/ copy clobbered the INFINC.006 deploy to
    # code/manual_api/ for 3 days, undetected. This sync now DETECTS basename
    # collisions and FAILS LOUD on any with differing content.
    local sync_rc
    python3 - <<PYEOF
import boto3, hashlib, os, sys

bucket   = '${INF_S3_BUCKET}'
prefixes = ['code/gold/', 'code/manual_api/', 'code/silver/', 'code/bronze/']
dest     = '${SCRIPTS}'

# Declared precedence for INTENTIONALLY multi-homed basenames: basename ->
# ordered prefix list (highest-precedence first). Empty by default --
# ADVI_INFETL.SYNC-COLLISION Phase 0 found NO intentional multi-homing; every
# historical collision was accidental deploy cruft and was normalized to
# byte-identical copies in Phase 1. A future genuinely-shared script gets an
# explicit entry here; anything else that collides with DIFFERING content
# fails loud below instead of silently picking a last-write-wins coin flip.
DECLARED_PRECEDENCE = {}

# Python package markers legitimately recur across prefixes and carry no
# chain logic -- exempt from collision detection.
COLLISION_EXEMPT = {'__init__.py'}

os.makedirs(dest, exist_ok=True)
s3  = boto3.client('s3', region_name='us-east-1')
pag = s3.get_paginator('list_objects_v2')

# Pass 1 -- inventory every object keyed by basename (the flat-dir key).
inventory = {}
for prefix in prefixes:
    for page in pag.paginate(Bucket=bucket, Prefix=prefix):
        for obj in page.get('Contents', []):
            fname = obj['Key'].split('/')[-1]
            if not fname:
                continue
            body = s3.get_object(Bucket=bucket, Key=obj['Key'])['Body'].read()
            inventory.setdefault(fname, []).append({
                'prefix': prefix,
                'key': obj['Key'],
                'body': body,
                'sha': hashlib.sha256(body).hexdigest(),
                'size': obj['Size'],
                'mtime': obj['LastModified'].isoformat(),
            })

# Pass 2 -- a basename in 2+ prefixes with DIFFERING content would silently
# run whichever copy downloaded last. Detect it; fail loud unless declared.
collisions = []
for fname, copies in sorted(inventory.items()):
    if len(copies) < 2 or fname in COLLISION_EXEMPT:
        continue
    if len({c['sha'] for c in copies}) == 1:
        print(f'  collision OK (identical x{len(copies)}): {fname}')
        continue
    if fname in DECLARED_PRECEDENCE:
        print(f'  collision RESOLVED (declared precedence): {fname}')
        continue
    collisions.append((fname, copies))

if collisions:
    sys.stderr.write(
        '\nFATAL: sync_scripts() basename collision -- the same script name '
        'exists in 2+ code/ prefixes with DIFFERENT content. A flat-dir sync '
        'would silently run whichever copy downloaded last. Refusing to sync.\n'
    )
    for fname, copies in collisions:
        sys.stderr.write(f'  COLLISION: {fname}\n')
        for c in copies:
            sys.stderr.write(
                f'    {c["key"]}  size={c["size"]}  mtime={c["mtime"]}  '
                f'sha256={c["sha"][:16]}\n'
            )
    sys.stderr.write(
        'Resolution: re-deploy the canonical copy over every prefix so all '
        'copies are byte-identical, OR add a DECLARED_PRECEDENCE entry if the '
        'multi-homing is intentional.\n'
    )
    sys.exit(1)

# Pass 3 -- write one copy per basename. Declared-precedence collisions take
# the highest-precedence prefix; everything else is collision-free or proven
# byte-identical, so any copy is correct.
n = 0
for fname, copies in inventory.items():
    chosen = copies[0]
    order = DECLARED_PRECEDENCE.get(fname)
    if order:
        for pfx in order:
            match = [c for c in copies if c['prefix'] == pfx]
            if match:
                chosen = match[0]
                break
    with open(os.path.join(dest, fname), 'wb') as fh:
        fh.write(chosen['body'])
    n += 1
print(f'Synced {n} files to {dest}')
PYEOF
    sync_rc=$?
    if [ "$sync_rc" -ne 0 ]; then
        echo "FATAL: sync_scripts() aborted (rc=$sync_rc) -- basename collision or S3 error. The chain will not run on stale/ambiguous code; see above." >&2
        exit 1
    fi
    echo "Synced: $(ls "$SCRIPTS"/*.py 2>/dev/null | wc -l) scripts"

    # Pull gold_prof.py from canonical opal-runner-shared bucket (cross-pipeline
    # telemetry profiler). Overrides any legacy copy from ${INF_S3_BUCKET}/code/gold/
    # in the sync above. Safe-fallback: if pull fails, the legacy copy remains.
    # ADHC 2026-05-13.
    if aws s3 cp s3://opal-runner-shared/code/gold/gold_prof.py "${SCRIPTS}/gold_prof.py" 2>/dev/null; then
        echo "gold_prof.py pulled from opal-runner-shared (canonical)"
    else
        echo "WARN: opal-runner-shared pull failed for gold_prof.py — using legacy copy from ${INF_S3_BUCKET}"
    fi
}

run_script() {
    local id="$1"

    local entry
    entry=$(echo "$CATALOG" | grep "^${id}|" || true)
    if [ -z "$entry" ]; then
        echo "ERROR: unknown script ID '$id'"
        echo "Valid IDs: $(echo "$CATALOG" | cut -d'|' -f1 | tr '\n' ' ')"
        exit 1
    fi

    local filename default_flags
    filename=$(echo "$entry" | cut -d'|' -f2)
    default_flags=$(echo "$entry" | cut -d'|' -f3)
    default_flags="${default_flags//\{date\}/$RUN_DATE}"

    if [ "$DRY_RUN" = "1" ]; then
        default_flags="${default_flags//--upload/--dry-run}"
    fi

    if [ ! -f "$SCRIPTS/$filename" ]; then
        echo "ERROR: script not found in S3 sync: $filename"
        echo "  Expected: $SCRIPTS/$filename"
        echo "  Upload to: s3://${INF_S3_BUCKET}/code/gold/$filename"
        exit 1
    fi

    local _marker_key="gold/artifacts/${id}/${RUN_DATE}"
    if [ "$SKIP_EXISTING" = "1" ]; then
        if aws s3api head-object --bucket "$INF_S3_BUCKET" --key "$_marker_key" >/dev/null 2>&1; then
            echo "SKIP: $id -- artifact exists for $RUN_DATE"
            return 0
        fi
    fi

    echo ""
    echo "========================================"
    echo "  $id  ($filename)"
    echo "  $(log_ts)"
    if [ -n "$default_flags" ]; then
        echo "  flags: $default_flags"
    fi
    echo "========================================"

    telem_step_begin "$id" 2>/dev/null || true

    # ADVI_BREAKS pull-durability (2026-06-30): the Orion REST dimension pulls
    # (INF_API_PULL_*) hang PERSISTENTLY with no server-side timeout --
    # pull_inflection_performance_rest.py hung 103min then 41min on Orion's perf
    # endpoint, blocking the whole chain under `set -e` (a kill aborts the chain
    # before D-stage). Mirror run_eclipse_step: bound each PULL with a timeout and
    # TOLERATE failure so a hung/slow pull fails fast and the chain still reaches
    # B_00 -> D. Safe by construction: these inline pulls are REDUNDANT re-pulls of
    # dims the scheduled cron Lambdas already land in today's bronze (12:30-13:40Z),
    # so a tolerated pull falls back to the cron's fresh bronze for that dim; and the
    # served AUA derives from positions/holdings, so even a stale perf still yields a
    # fresh-AUA rich header. NON-pull steps (silver/A/B/C/D) still hard-fail as before.
    local _is_pull=0
    case "$id" in
        INF_API_PULL_*) _is_pull=1 ;;
    esac

    local _run_rc=0
    if [ "$_is_pull" = "1" ]; then
        GOLD_CHAIN_LABEL="$CHAIN_LABEL" timeout -k 30 "${PULL_STEP_TIMEOUT:-600}" \
            python3 "$SCRIPTS/gold_prof.py" "$SCRIPTS/$filename" $default_flags
        _run_rc=$?
        if [ "$_run_rc" -ne 0 ]; then
            echo "  WARN: $id pull timed out/failed (exit=$_run_rc, timeout=${PULL_STEP_TIMEOUT:-600}s) -- TOLERATING; chain proceeds on prior/cron bronze for this dim (ADVI_BREAKS pull-durability)"
            telem_error "$id pull tolerated (exit=$_run_rc)" 2>/dev/null || true
            telem_step_done "$id" --output "$id pull tolerated (exit=$_run_rc)" --status warn 2>/dev/null || true
            return 0
        fi
    else
        GOLD_CHAIN_LABEL="$CHAIN_LABEL" python3 "$SCRIPTS/gold_prof.py" "$SCRIPTS/$filename" $default_flags
        _run_rc=$?
    fi

    if [ "$_run_rc" -eq 0 ]; then
        telem_step_done "$id" --output "$id complete" --status ok 2>/dev/null || true
        echo "  $id done"
    else
        telem_error "$id failed with exit=$_run_rc" 2>/dev/null || true
        telem_step_done "$id" --output "$id failed (exit=$_run_rc)" --status fail 2>/dev/null || true
        echo "  $id FAILED (exit=$_run_rc)"
        return "$_run_rc"
    fi

    if [ "$SKIP_EXISTING" = "1" ]; then
        aws s3 cp /dev/null "s3://${INF_S3_BUCKET}/${_marker_key}" --quiet 2>/dev/null || true
    fi
}

# ---------------------------------------------------------------------------
# run_eclipse_step -- BEST-EFFORT eclipse step runner (ADVI_ECLDAT, Andrew
# directive 2026-05-15). Used ONLY by the post-sentinel eclipse block. Unlike
# run_script:
#   - timeout-bounded (ECLIPSE_STEP_TIMEOUT, default 600s) so a hung eclipse
#     step cannot stall the runner / delay auto-shutdown.
#   - NEVER returns non-zero -- an eclipse failure must not abort the chain or
#     change chain_exit. Failures + timeouts are logged + recorded in
#     ECLIPSE_RESULTS for the observability marker.
# ---------------------------------------------------------------------------
ECLIPSE_STEP_TIMEOUT="${ECLIPSE_STEP_TIMEOUT:-600}"
ECLIPSE_RESULTS=""

run_eclipse_step() {
    local id="$1"
    local entry filename default_flags
    entry=$(echo "$CATALOG" | grep "^${id}|" || true)
    if [ -z "$entry" ]; then
        echo "  SKIP: eclipse step '$id' not in CATALOG"
        ECLIPSE_RESULTS="$ECLIPSE_RESULTS $id:missing"
        return 0
    fi
    filename=$(echo "$entry" | cut -d'|' -f2)
    default_flags=$(echo "$entry" | cut -d'|' -f3)
    default_flags="${default_flags//\{date\}/$RUN_DATE}"
    if [ "$DRY_RUN" = "1" ]; then
        default_flags="${default_flags//--upload/--dry-run}"
    fi
    if [ ! -f "$SCRIPTS/$filename" ]; then
        echo "  WARN: eclipse script not found in sync: $filename (skipping, chain unaffected)"
        ECLIPSE_RESULTS="$ECLIPSE_RESULTS $id:missing"
        return 0
    fi

    echo ""
    echo "  --- $id ($filename) $(log_ts) ---"
    if [ -n "$default_flags" ]; then
        echo "  flags: $default_flags"
    fi

    local rc=0
    timeout -k 60 "$ECLIPSE_STEP_TIMEOUT" \
        python3 "$SCRIPTS/gold_prof.py" "$SCRIPTS/$filename" $default_flags || rc=$?

    if [ "$rc" -eq 0 ]; then
        echo "  $id done (eclipse best-effort)"
        ECLIPSE_RESULTS="$ECLIPSE_RESULTS $id:ok"
    elif [ "$rc" -eq 124 ] || [ "$rc" -eq 137 ]; then
        echo "  WARN: $id TIMED OUT after ${ECLIPSE_STEP_TIMEOUT}s -- eclipse best-effort, chain UNAFFECTED"
        ECLIPSE_RESULTS="$ECLIPSE_RESULTS $id:timeout"
    else
        echo "  WARN: $id FAILED exit=$rc -- eclipse best-effort, chain UNAFFECTED"
        ECLIPSE_RESULTS="$ECLIPSE_RESULTS $id:fail($rc)"
    fi
    # BRONZE-FIX #7 AC5: best-effort CANNOT fatal by design (COO~B ruling),
    # so a failed/timed-out eclipse step gets its own metric instead -- a
    # silently-dark eclipse leg froze Model Portfolios in the 06-12 P0 (L4).
    # (Same IAM-scoped namespace; no alarm yet -- alarm spec rides AE wiring.)
    if [ "$rc" -ne 0 ]; then
        aws cloudwatch put-metric-data --namespace "$ALERT_CW_NAMESPACE" \
            --metric-name EclipseStepFailed --value 1 --unit Count \
            --dimensions "Service=$ALERT_CW_SERVICE,Step=$id" 2>/dev/null || true
    fi
    return 0
}

# ---------------------------------------------------------------------------
# step_is_date_capable -- ADVI_INFETL.RUNDATE-DISPATCH-IMPL E1 (CTO RATIFY
# 20260515T193422Z, Element 1 = Option 1b). A catalog step is date-capable iff
# its default_flags field carries the literal {date} token (the only thing
# --run-date can substitute into). Date-agnostic steps have no {date} token --
# they ignore --run-date and pull current/full-history data. See the
# DATE-CAPABILITY CONTRACT comment at the CATALOG block.
# Returns 0 = date-capable, 1 = date-agnostic (also 1 for an unknown ID, so the
# guard errs toward rejecting --run-date -- never toward a silent mixed-date run).
# ---------------------------------------------------------------------------
step_is_date_capable() {
    local id="$1"
    local entry flags
    entry=$(echo "$CATALOG" | grep "^${id}|" || true)
    if [ -z "$entry" ]; then
        return 1
    fi
    flags=$(echo "$entry" | cut -d'|' -f3)
    case "$flags" in
        *'{date}'*) return 0 ;;
        *)          return 1 ;;
    esac
}

# ---------------------------------------------------------------------------
# validate_rundate_coherence -- ADVI_INFETL.RUNDATE-DISPATCH-IMPL E2. The
# reject-incoherent-combination guard. --run-date shifts the single global
# RUN_DATE, but only date-capable steps honor it; date-agnostic steps ignore it
# and pull current/full-history data. A resolved chain that mixes the two under
# --run-date is a category error -> a mixed-date artifact (or, more usually, a
# bronze-not-found failure). Accept --run-date ONLY if every resolved step is
# date-capable; else REJECT loudly (exit non-zero), naming the offending
# date-agnostic step(s). Active only when RUN_DATE_OVERRIDE is non-empty.
# Evaluated after chain resolution, before run_chain / the --bg re-invoke -- the
# single dispatcher chokepoint catching the inf-gold-trigger Lambda, SSM-direct
# dispatch, and manual CLI alike.
# ---------------------------------------------------------------------------
validate_rundate_coherence() {
    [ -z "$RUN_DATE_OVERRIDE" ] && return 0
    local id agnostic=""
    for id in $SCRIPT_IDS; do
        if ! step_is_date_capable "$id"; then
            agnostic="$agnostic $id"
        fi
    done
    agnostic="${agnostic# }"
    if [ -n "$agnostic" ]; then
        echo "ERROR: --run-date $RUN_DATE_OVERRIDE rejected -- incoherent step combination."
        echo "  The resolved chain contains date-agnostic step(s): $agnostic"
        echo "  Date-agnostic steps pull current/full-history data and ignore --run-date,"
        echo "  while date-capable steps would be pinned to $RUN_DATE_OVERRIDE -- this produces a mixed-date artifact."
        echo "  --run-date is valid only for a chain composed entirely of date-capable steps"
        echo "  (e.g. --single INF_API_SILVER_TXN --run-date $RUN_DATE_OVERRIDE, against an existing bronze pull)."
        echo "  For a normal full run, use --chain full with no --run-date."
        exit 1
    fi
}

run_chain() {
    local chain_name="$1"
    local script_ids="$2"

    echo "================================================================"
    echo "  CHAIN: $chain_name"
    echo "  Started: $(log_ts)"
    echo "  Scripts: $script_ids"
    echo "================================================================"

    # BRONZE-FIX #7: arm the FATAL loud channel BEFORE sync_scripts -- the
    # 2026-06-12 13:00Z silent outage died INSIDE sync_scripts; arming after
    # it would miss the exact failure class this exists for.
    CHAIN_STARTED=1

    sync_scripts

    # Run ID format matches BD precedent: {chain-label}_{YYYYMMDD_HHMMSS}
    # Computed unconditionally so INFINC_CHAIN_RUN_ID is exported into every
    # child run_script invocation — partition_provenance.collect() reads it via
    # os.environ.get('INFINC_CHAIN_RUN_ID') for forensic audit-trail in every
    # Pattern A+B partition write (DS~D d88b7a7a4 contract, RAVC2 gap-fix).
    local _telem_run_id
    _telem_run_id="${chain_name//[^A-Za-z0-9_-]/_}_$(date -u +%Y%m%d_%H%M%S)"
    export INFINC_CHAIN_RUN_ID="$_telem_run_id"

    # Telemetry: init + start 30s push loop (gated on TELEM_AVAILABLE)
    if [ "$TELEM_AVAILABLE" = "1" ]; then
        local _telem_mode _telem_step_count _telem_step_names
        _telem_mode="${chain_name#chain:}"  # strip "chain:" prefix to get full/incremental/gold-d/etc.
        _telem_step_count=$(echo "$script_ids" | wc -w | tr -d ' ')
        _telem_step_names=$(echo "$script_ids" | tr ' ' ',')
        telem_init "$TELEM_SERVICE" "$_telem_run_id" \
            --mode "$_telem_mode" \
            --steps "$_telem_step_count" \
            --step-names "$_telem_step_names" \
            --s3-bucket "$INF_S3_BUCKET" 2>/dev/null || true
        # Background 30s push loop.
        # CRITICAL: close stdin/stdout/stderr in the subshell so it doesn't
        # inherit run_chain's pipe-to-tee write end. If we leave stdout open,
        # the chain-wrap-up `run_chain | tee log` in main() deadlocks: this
        # forever-loop never closes stdout, so tee never gets EOF, so main()
        # waits forever for tee to exit. Empirical: chain #28 hung 10min in
        # do_wait until SSM-killed (2026-05-13). Redirect all 3 FDs to /dev/null.
        (
            exec </dev/null >/dev/null 2>&1
            while true; do
                sleep 30
                bash "$SCRIPTS/telemetry-sync.sh" || true
            done
        ) &
        TELEM_SYNC_PID=$!
        disown "$TELEM_SYNC_PID" 2>/dev/null || true
    fi

    local _chain_rc=0
    for id in $script_ids; do
        # --- ADHC_VALGATE.001 Phase 4 -- Class-C data-correctness gate ---
        # Fire the C-1 (row-count) + C-2 (classification-distribution) gate
        # post-D-chain / pre-platinum-emit: immediately before INF_09, the
        # portal/platinum emit. v1 is WARN-not-BLOCK (PRD section 7-Q1): the
        # gate's [VALGATE-C] verdict lines surface to the chain log; it does
        # NOT block the chain. NON-FATAL by construction (COO~B ruling
        # 2026-05-20, option a) -- a non-zero exit is swallowed with a WARN.
        # Rationale: a v1 WARN-only gate, and especially its own infra error,
        # must never fail the platinum emit. valgate_gate.py is not yet
        # crash-safe (an infra error exits non-zero, ambiguous with a real
        # BLOCK verdict), so a literal exit-code-gated wire is unsafe in v1.
        # The WARN->BLOCK ratchet is a 3-change sequence (GATE_VERDICT_POLICY
        # flip + valgate_gate.py crash-safety [HARD prerequisite] + this wire
        # flipped to fatal) -- see the ADHC_VALGATE.001 ratchet-readiness
        # checklist.
        if [ "$id" = "INF_09" ]; then
            echo ""
            echo "  [VALGATE-C] Class-C gate (C-1 row-count / C-2 distribution / C-3 sentinel-rows / C-4 cross-source parity) -- silver/transactions @ ${RUN_DATE}, pre-platinum-emit"
            if [ -f "$SCRIPTS/valgate_gate.py" ]; then
                python3 "$SCRIPTS/valgate_gate.py" --artifact silver/transactions --date "$RUN_DATE" \
                    || echo "  [VALGATE-C] WARN: gate exited non-zero -- v1 is WARN-not-BLOCK; non-fatal, chain continues. Investigate the gate (infra error, or -- post-ratchet -- a BLOCK verdict)."
            else
                echo "  [VALGATE-C] WARN: valgate_gate.py absent from $SCRIPTS -- Class-C gate skipped (non-fatal)."
            fi
        fi
        # CRITICAL: do NOT use `if ! run_script`. The `!` operator inverts
        # the exit code, so `$?` after captures 0 (the inverted code), not
        # run_script's real failure code. Chain #29 INF_D_00c (2026-05-13)
        # failed with exit=1 but logged "FAILED at INF_D_00c (exit=0)" and
        # wrote pipeline-complete.json (false-positive success sentinel),
        # which auto-fired the publisher Lambda on incomplete data.
        # Use `|| rc=$?` pattern to capture the real exit code.
        local _script_rc=0
        run_script "$id" || _script_rc=$?
        if [ "$_script_rc" -ne 0 ]; then
            _chain_rc=$_script_rc
            echo ""
            echo "================================================================"
            echo "  CHAIN $chain_name FAILED at $id (exit=$_chain_rc)"
            echo "  Aborted: $(log_ts)"
            echo "================================================================"
            return "$_chain_rc"
        fi
        # RE-PIN THE AoR SOURCE THE MOMENT THE PULL REWRITES IT (round-8.2). The pre-chain pin was taken
        # BEFORE this script ran, so it names the partition this script just superseded (first run of the
        # day) or OVERWROTE IN PLACE (any same-day re-run). Left alone, the first case silently reads a
        # day-stale roster and the second RAISES on a healthy nightly and kills the whole chain. Pin what
        # the resolver will ACTUALLY read.
        if [ "$id" = "INF_API_SILVER_ACCOUNTS_V2" ]; then
            _repin_aor_source_after_pull
        fi
    done

    echo ""
    echo "================================================================"
    echo "  CHAIN $chain_name COMPLETE"
    echo "  Finished: $(log_ts)"
    echo "================================================================"
}

# ---------------------------------------------------------------------------
# check_partial_deps -- ADHC_COMPIL.002
# For each script in the partial chain, calls comp-pipescript postdeps and
# checks that all [BLOCKING] Must run: dependencies are also in the chain.
# Returns 0 if satisfied; prints error + suggested command and returns 1 if not.
# Skips gracefully (returns 0) if comp-pipescript is not in PATH.
# ---------------------------------------------------------------------------
check_partial_deps() {
    local partial_ids="$1"

    if ! command -v comp-pipescript >/dev/null 2>&1; then
        echo "  WARN: comp-pipescript not found -- skipping postdeps gate."
        return 0
    fi

    local missing=""
    for id in $partial_ids; do
        local postdeps_out
        postdeps_out=$(comp-pipescript postdeps "$id" 2>/dev/null || true)

        while IFS= read -r line; do
            if echo "$line" | grep -q '\[BLOCKING\] Must run:'; then
                local dep_file
                dep_file=$(echo "$line" | sed 's/.*\[BLOCKING\] Must run: *//')
                local dep_id
                dep_id=$(echo "$CATALOG" | awk -F'|' -v fn="$dep_file" '$2 == fn {print $1; exit}')
                if [ -n "$dep_id" ] && ! echo " $partial_ids " | grep -q " $dep_id "; then
                    if ! echo " $missing " | grep -q " $dep_id "; then
                        missing="$missing $dep_id"
                    fi
                fi
            fi
        done <<< "$postdeps_out"
    done

    missing="${missing# }"
    if [ -z "$missing" ]; then
        return 0
    fi

    echo "ERROR: --chain partial missing required post-step scripts:"
    local all_needed="$partial_ids $missing"
    for m in $missing; do
        echo "  $m"
    done
    echo ""
    local ordered=""
    for id in $CHAIN_FULL; do
        if echo " $all_needed " | grep -q " $id "; then
            ordered="$ordered $id"
        fi
    done
    echo "Suggested corrected command:"
    echo "  bash $(basename "$0") --chain partial${ordered}"
    return 1
}

require_arg() {
    local flag="$1"
    local val="$2"
    if [ -z "$val" ] || [[ "$val" == --* ]]; then
        echo "ERROR: $flag requires an argument (got '${val:-<empty>}')."
        echo "Run with --help for usage."
        exit 1
    fi
}

while [ $# -gt 0 ]; do
    case "$1" in
        --chain)
            require_arg "$1" "${2:-}"; MODE="chain"; TARGET="$2"; shift 2
            if [ "$TARGET" = "partial" ]; then
                while [ $# -gt 0 ] && [[ "$1" != --* ]]; do
                    PARTIAL_IDS="$PARTIAL_IDS $1"; shift
                done
                PARTIAL_IDS="${PARTIAL_IDS# }"
                if [ -z "$PARTIAL_IDS" ]; then
                    echo "ERROR: --chain partial requires at least one script ID."
                    echo "  Example: --chain partial INF_B_03 INF_C_00"
                    exit 1
                fi
            fi
            ;;
        --single)   require_arg "$1" "${2:-}"; MODE="single"; TARGET="$2"; shift 2 ;;
        --from)     require_arg "$1" "${2:-}"; MODE="from";   TARGET="$2"; shift 2 ;;
        --stage)    require_arg "$1" "${2:-}"; MODE="stage";  TARGET="$2"; shift 2 ;;  # ADVI_INFREF P1 LEG-2: pull | compute (separately-invocable PULL/COMPUTE module)
        --bg)          BG=1;          shift ;;
        --dry-run)     DRY_RUN=1;     shift ;;
        --no-shutdown) NO_SHUTDOWN=1; shift ;;
        --no-deps-check) NO_DEPS_CHECK=1; shift ;;
        --no-freshness-check) NO_FRESHNESS_CHECK=1; shift ;;
        --bypass-reason) require_arg "$1" "${2:-}"; BYPASS_REASON="$2"; shift 2 ;;
        --force)       FORCE=1;       shift ;;
        --run-date) require_arg "$1" "${2:-}"; RUN_DATE_OVERRIDE="$2"; shift 2 ;;
        --rebuild)              REBUILD=1; shift ;;
        --rebuild-start-date)   require_arg "$1" "${2:-}"; REBUILD_START="$2"; shift 2 ;;
        --rebuild-end-date)     require_arg "$1" "${2:-}"; REBUILD_END="$2"; shift 2 ;;
        --rebuild-advisor-attribution-only) REBUILD_ADVISOR_ATTRIBUTION_ONLY=1; shift ;;
        --skip-existing) SKIP_EXISTING=1; shift ;;
        --status)   do_status; exit 0 ;;
        --kill)     do_kill;   exit 0 ;;
        --cancel-shutdown)
            # BRONZE-FIX #9 AC4: first-class IDEMPOTENT cancel (what DS did
            # ad-hoc at 21:27Z during the P0 recovery). Safe to run whether or
            # not a shutdown is pending.
            if sudo shutdown -c 2>/dev/null; then
                echo "SHUTDOWN-CANCELLED instance=$(hostname) $(date -u +%Y-%m-%dT%H:%M:%SZ)"
            else
                echo "NO-SHUTDOWN-PENDING instance=$(hostname) $(date -u +%Y-%m-%dT%H:%M:%SZ) (cancel is idempotent)"
            fi
            exit 0 ;;
        --help|-h)  show_help; exit 0 ;;

# INF_AOR_SRC_ETAG -- THE SOURCE CONTENT PIN (round-8.2, CTO ruling 2026-07-13).
#
# INF_AOR_ASOF above is the partition DATE. It locates the object and it is a useful human-readable
# vintage -- but it is NOT an identity and NOTHING asserts on it. The partition is a single fixed key
# (snapshot_date=<d>/data.parquet), so a re-emit OVERWRITES CONTENT IN PLACE and the date REPEATS ACROSS
# DIFFERENT BYTES: 6 of 69 accounts_v2 objects were rewritten AFTER their own snapshot_date (all on
# 2026-05-11, a backfill). A date-keyed guard cannot see that, and would pass a resolution derived from
# content that no longer exists.
#
# The resolution's identity is (source_digest, fold_mode, code_digest) -- the CONTENT it came from, the
# SEMANTICS it was derived under, and the CODE that derived it. INF_A_0 stamps all three;
# load_resolution() asserts all three (lib/aor_resolution.py).
#
# PIN THE CONTENT DIGEST ONCE, HERE, AND EXPORT IT -- consumers compare against this pin and NEVER HEAD S3
# themselves. load_resolution() runs inside per-snapshot / 32-worker loops, so a live HEAD per call would
# cost a round-trip per worker AND make the guard NON-DETERMINISTIC: a partition rewritten mid-chain would
# make one consumer pass and the next raise, in the SAME run, with no bug present. An instrument whose
# verdict depends on when you asked it is not an instrument. Pin once; compare many.
# (The resolver stamps the ETag of the object it ACTUALLY READ -- taken off the GET that produced the
# frame -- and asserts it against this pin ONCE. If the loader walked back, that is a real condition and
# it is loud.)
#
# A run IDENTITY was considered and WITHDRAWN: it over-fires. The resolution is a pure function of its
# inputs, so a prior run's artifact with identical inputs is a CORRECT CACHE, not staleness -- run-id
# would hard-abort `--single INF_D_00c`, the documented example in this script's own help text.
if [ -z "${INF_AOR_SRC_ETAG:-}" ] && [ -n "${INF_AOR_ASOF:-}" ]; then
    _aor_etag="$(aws s3api head-object --bucket "$INF_S3_BUCKET" \
        --key "silver/api_dims/accounts_v2/snapshot_date=${INF_AOR_ASOF}/data.parquet" \
        --query 'ETag' --output text 2>/dev/null | tr -d '"' || true)"
    if [ -n "$_aor_etag" ] && [ "$_aor_etag" != "None" ]; then
        export INF_AOR_SRC_ETAG="$_aor_etag"
        echo "INFO: INF_AOR_SRC_ETAG pinned to $INF_AOR_SRC_ETAG (advisor-of-record SOURCE CONTENT digest -- the identity the resolution is asserted against)."
    else
        echo "WARN: could not HEAD the accounts_v2 partition for INF_AOR_SRC_ETAG -- leaving UNSET. The AoR resolver/consumers will FAIL CLOSED (they refuse to skip the content check rather than fabricate one)."
    fi
fi

# =============================================================================================
# >> RE-PIN AFTER THE PULL. THIS IS NOT AN OPTIMISATION -- IT IS WHAT KEEPS THE GUARD FROM BECOMING
# >> AN OUTAGE, AND WHAT KEEPS THE ROSTER FROM BEING A DAY STALE. (round-8.2, CVO~B/CTO)
#
# THE PIN ABOVE IS TAKEN AT SCRIPT LOAD -- i.e. BEFORE `CHAIN_PULL` RUNS. And CHAIN_PULL contains
# INF_API_SILVER_ACCOUNTS_V2, which WRITES silver/api_dims/accounts_v2/snapshot_date=$RUN_DATE. So the
# thing we pinned is the thing the pull is about to supersede or overwrite. Both branches are broken, and
# they are broken in OPPOSITE directions:
#
#   FIRST FULL RUN OF THE DAY -- today's partition does not exist yet, so `s3 ls | tail -1` pins
#     YESTERDAY. The pull then writes TODAY. The resolver (and A_00/A_01/C_07, which now honour the pin)
#     all read YESTERDAY'S ROSTER while today's fresh one sits there unused. Silent staleness -- and it is
#     round-8.2's F2 fix that dragged the consumers onto the stale pin (before it, they walked back to the
#     LATEST partition and read today's). Fixing the skew by aligning everyone onto the WRONG partition is
#     not fixing it.
#
#   ANY SAME-DAY RE-RUN (a recovery run -- exactly when this matters most) -- today's partition EXISTS, so
#     we pin it and take its ETag E1. The pull then REWRITES that same fixed key -> E2. The resolver reads
#     it, stamps E2, and asserts E2 == the pin E1 -> MISMATCH -> RAISE -> INF_A_0 is FIRST in the chain ->
#     THE ENTIRE NIGHTLY DIES. A guard built to stop silent corruption becomes a TOTAL OUTAGE THAT READS
#     AS A CORRECT FAIL-CLOSED. That is the C_07 lesson inverted: an instrument that fires on the healthy
#     case is not a guard, it is an outage with a good explanation.
#
# THE FIX IS THE ORDERING, NOT THE GUARD: PIN WHAT THE RESOLVER WILL ACTUALLY READ. Re-pin immediately
# after the accounts_v2 silver rebuild lands, so the pin names the partition that now exists. Both branches
# then collapse to the same correct behaviour: the pin, the resolver's stamp, and every consumer's
# expectation all name the SAME bytes -- the ones the pull just wrote.
#
# An OPERATOR-SUPPLIED pin is never clobbered: re-pinning applies only when we derived the pin ourselves.
# =============================================================================================
if [ -z "${INF_AOR_PIN_OPERATOR:-}" ]; then
    INF_AOR_PIN_AUTO=1
fi

_repin_aor_source_after_pull() {
    # Called from run_chain immediately after INF_API_SILVER_ACCOUNTS_V2 succeeds.
    [ "${INF_AOR_PIN_AUTO:-0}" = "1" ] || return 0
    local _d _e
    _d="$(aws s3 ls "s3://${INF_S3_BUCKET}/silver/api_dims/accounts_v2/" 2>/dev/null \
        | grep -oE 'snapshot_date=[0-9]{4}-[0-9]{2}-[0-9]{2}' | sort | tail -1 | sed 's/snapshot_date=//' || true)"
    if ! printf '%s' "$_d" | grep -qE '^[0-9]{4}-[0-9]{2}-[0-9]{2}$'; then
        echo "WARN: RE-PIN after the accounts_v2 pull could not resolve a partition -- the AoR guard will FAIL CLOSED rather than assert against a pre-pull pin."
        unset INF_AOR_ASOF INF_AOR_SRC_ETAG
        return 0
    fi
    _e="$(aws s3api head-object --bucket "$INF_S3_BUCKET" \
        --key "silver/api_dims/accounts_v2/snapshot_date=${_d}/data.parquet" \
        --query 'ETag' --output text 2>/dev/null | tr -d '"' || true)"
    if [ -z "$_e" ] || [ "$_e" = "None" ]; then
        echo "WARN: RE-PIN after the accounts_v2 pull could not HEAD snapshot_date=${_d} -- unsetting the pin so the AoR guard FAILS CLOSED rather than asserting against stale bytes."
        unset INF_AOR_ASOF INF_AOR_SRC_ETAG
        return 0
    fi
    if [ "${INF_AOR_ASOF:-}" != "$_d" ] || [ "${INF_AOR_SRC_ETAG:-}" != "$_e" ]; then
        echo "INFO: RE-PINNED the AoR source AFTER the accounts_v2 pull -- as-of ${INF_AOR_ASOF:-<unset>} -> ${_d}, digest ${INF_AOR_SRC_ETAG:-<unset>} -> ${_e}. The pre-pull pin named the partition the pull was about to supersede/overwrite; the resolver and every consumer now assert against the bytes the pull ACTUALLY wrote."
    fi
    export INF_AOR_ASOF="$_d"
    export INF_AOR_SRC_ETAG="$_e"
}

# NFRA_INFRPT.GUARD-REFILE leg-B (0.7.0_AE_CloudOps, 2026-07-08): re-arm the
# watchdog-shutdown timer via an EXIT trap as soon as a real chain-bearing
# dispatch commits (this point is only reached past the argument-parsing
# loop's --status/--kill/--cancel-shutdown/--help exits, so CHAIN_LABEL is
# always set here) -- closing #9-F1 (a crashed dispatch that never reaches
# the Layer-1 arm block below previously left the instance with NO shutdown
# path at all, since run-gold-layer.sh DELETES the watchdog unit files at
# boot). "Re-arm" here means RECREATE the unit files (identical content to
# user-data.sh's bootstrap block) since the files are gone, not merely
# stopped, then enable+start the timer. Safe on a normal completion: the
# Layer-1 `shutdown -h +2` below fires first (2min grace) and the instance
# is down long before the watchdog's own 5min cycle + 300s grace could act;
# the watchdog script's OWN pgrep-based activity check (Check 1) also skips
# any cycle where a gold process is genuinely still running, so an early
# re-arm cannot kill an in-flight chain (mirrors the #9 concurrency-aware
# guard's own reasoning below).
# ---------------------------------------------------------------------------
_rearm_watchdog_timer() {
    sudo tee /etc/systemd/system/watchdog-shutdown.service >/dev/null << 'SVC' || true
[Unit]
Description=Gold Runner Watchdog Shutdown

[Service]
Type=oneshot
ExecStart=/usr/local/bin/watchdog-shutdown.sh
SVC
    sudo tee /etc/systemd/system/watchdog-shutdown.timer >/dev/null << 'TMR' || true
[Unit]
Description=Run watchdog every 5 minutes

[Timer]
OnBootSec=10min
OnUnitActiveSec=5min

[Install]
WantedBy=timers.target
TMR
    sudo systemctl daemon-reload 2>/dev/null || true
    sudo systemctl enable watchdog-shutdown.timer 2>/dev/null || true
    sudo systemctl start watchdog-shutdown.timer 2>/dev/null || true
    echo "[$(date -u +%Y-%m-%dT%H:%M:%SZ)] WATCHDOG_REARMED chain=$CHAIN_LABEL -- crash-path auto-reap net restored (#9-F1 closed)."
    # GLDRNS.B1 forensics hotfix: ship the newest chain log to S3 on EVERY exit -- crash or
    # complete -- so a death is never unreadable again. || true throughout: a trap must not fail.
    _newest_log=$(ls -t "$LOGDIR"/*.log 2>/dev/null | head -1) || true
    if [ -n "$_newest_log" ]; then
        aws s3 cp "$_newest_log" "s3://${INF_S3_BUCKET}/diagnostic/chain_logs/$(basename "$_newest_log")" 2>/dev/null || true
        echo "[$(date -u +%Y-%m-%dT%H:%M:%SZ)] EXIT-TRAP log shipped: $(basename "$_newest_log") -> s3 diagnostic/chain_logs/" || true
    fi
}

# Exit 0 (PASS/WARN): proceed. Exit 1 (FAIL): block + print remediation.

# Sequence: deployed check -> .002 cascade gate -> .013 freshness gate -> dispatch.
# ---------------------------------------------------------------------------
if [ "$MODE" = "chain" ]; then
    if [ "$NO_FRESHNESS_CHECK" = "1" ]; then
        log_freshness_bypass_entry "$(basename "$0")" "$CHAIN_LABEL" "$BYPASS_REASON"
    elif ! command -v comp-freshness >/dev/null 2>&1; then
        echo "  WARN: comp-freshness not found in PATH -- freshness gate skipped."
    else
        echo "=== comp-freshness gate ($(log_ts)) ==="
        _freshness_exit=0
        comp-freshness chain inf || _freshness_exit=$?
        if [ "$_freshness_exit" = "1" ]; then
            echo ""
            echo "DISPATCH BLOCKED: comp-freshness FAIL -- one or more input artifacts are stale."
            echo "  Next steps:"
            echo "    1. Re-run the upstream silver/bronze backfill for the failing artifact(s)."
            echo "    2. If an Orion API dim feed, check the INF bronze pull scripts and API status."
            echo "    3. Once inputs are fresh, re-dispatch normally."
            echo "  Emergency bypass (planned outage / operator override):"
            echo "    Add: --no-freshness-check [--bypass-reason \"<reason>\"]"
            exit 1
        elif [ "$_freshness_exit" = "2" ]; then
            echo ""
            echo "DISPATCH BLOCKED: comp-freshness config error or AWS unreachable (exit 2)."
            echo "  This is a tooling failure -- NOT a data-staleness FAIL."
            echo "  Remediate: verify AWS credentials, check cadence config YAML, then retry."
            echo "  To bypass during infra repair:"
            echo "    Add: --no-freshness-check --bypass-reason \"infra repair in progress\""
            exit 2
        fi
        echo "  comp-freshness gate: PASS (exit 0)"
    fi
fi

# ---------------------------------------------------------------------------
# Execute chain
# ---------------------------------------------------------------------------
chain_exit=0
if [ "$_in_bg_child" = "1" ]; then
    # `|| chain_exit=$?` (the :983 house pattern), NOT a bare call + `$?`: under
    # set -e an unguarded run_chain failure kills the script HERE, so the
    # sentinel block below (pipeline-failed.json) never runs. That is how the
    # 2026-07-17 14:53Z INF_C_01 guard abort died with no failure stamp -- the
    # bg child took this branch; the tee branch below survives because the
    # pipeline exits with tee's code.
    run_chain "$CHAIN_LABEL" "$SCRIPT_IDS" || chain_exit=$?
else
    run_chain "$CHAIN_LABEL" "$SCRIPT_IDS" 2>&1 | tee "$log_file"
    chain_exit=${PIPESTATUS[0]}
fi

# Telemetry: emit terminal status before background sync push.
# telem_complete sets status=complete; telem_error sets status=failed.
# trap _telem_cleanup on EXIT will fire the final sync push to S3 archive.
if [ "$TELEM_AVAILABLE" = "1" ]; then
    if [ "$chain_exit" -eq 0 ]; then
        telem_complete 2>/dev/null || true
    else
        telem_error "chain exit code $chain_exit" 2>/dev/null || true
    fi
fi

merge_perf_report

# Write gold/meta/pull-complete.json at --stage pull success. It is the assert
# target that makes --stage compute safely separately-invocable (compute asserts
# sentinel.run_date == its own RUN_DATE). Not written for dry-run / other modes.

case "$MODE:$TARGET" in

esac
if [ "$_ran_pull_stage" = "1" ] && [ "$DRY_RUN" = "0" ] && [ "${chain_exit:-1}" -eq 0 ]; then

        | aws s3 cp - "s3://${INF_S3_BUCKET}/gold/meta/pull-complete.json" \
            --content-type application/json; then

    else

    fi

# ---------------------------------------------------------------------------
# Chain-complete sentinel -- ADVI_INFETL.002
# Write gold/meta/pipeline-complete.json at --chain full success only.
# Downstream S3 ObjectCreated rule (TO-5) fires inf-portal-publisher on this key.
# Not written for dry-run, --single, --from, or --chain incremental.
# ---------------------------------------------------------------------------
if [ "$MODE" = "chain" ] && [ "$TARGET" = "full" ] && [ "$DRY_RUN" = "0" ]; then
    _sent_ts=$(date -u +%Y-%m-%dT%H:%M:%SZ)
    _sent_sha=$(git rev-parse --short HEAD 2>/dev/null || echo "unknown")
    _sent_imds_tok=$(curl -s -m2 -X PUT "http://169.254.169.254/latest/api/token" \
        -H "X-aws-ec2-metadata-token-ttl-seconds: 60" 2>/dev/null || true)
    _sent_iid=$(curl -s -m2 \
        -H "X-aws-ec2-metadata-token: $_sent_imds_tok" \
        "http://169.254.169.254/latest/meta-data/instance-id" 2>/dev/null || echo "unknown")
    if [ "$chain_exit" -eq 0 ]; then
        _sent_json=$(printf '{"chain":"full","run_date":"%s","completed_at":"%s","git_sha":"%s","instance_id":"%s"}' \
            "$RUN_DATE" "$_sent_ts" "$_sent_sha" "$_sent_iid")
        if echo "$_sent_json" \
            | aws s3 cp - "s3://${INF_S3_BUCKET}/gold/meta/pipeline-complete.json" \
                --content-type application/json; then
            echo "[$(date -u +%Y-%m-%dT%H:%M:%SZ)] [dispatch] Sentinel OK: s3://${INF_S3_BUCKET}/gold/meta/pipeline-complete.json written ($RUN_DATE)"
        else
            _sent_rc=$?
            echo "[$(date -u +%Y-%m-%dT%H:%M:%SZ)] [dispatch] WARN: sentinel write FAILED with exit=$_sent_rc -- chain completion signal not delivered (publisher trigger may not fire)" >&2
        fi
    else
        _fail_json=$(printf '{"chain":"full","run_date":"%s","failed_at":"%s","exit_code":%s,"git_sha":"%s","instance_id":"%s"}' \
            "$RUN_DATE" "$_sent_ts" "$chain_exit" "$_sent_sha" "$_sent_iid")
        echo "$_fail_json" \
            | aws s3 cp - "s3://${INF_S3_BUCKET}/gold/meta/pipeline-failed.json" \
                --content-type application/json 2>/dev/null || true
        echo "[$(date -u +%Y-%m-%dT%H:%M:%SZ)] [dispatch] Chain failed (exit=$chain_exit); NOT writing pipeline-complete.json sentinel (pipeline-failed.json written)"
    fi
fi

    fi

# Eclipse model-portfolios -- BEST-EFFORT, POST-SENTINEL block (ADVI_ECLDAT).
# Andrew directive 2026-05-15: "we cannot pause portal updates if eclipse has
# an issue, that is absolutely unacceptable." This block runs AFTER
# gold/meta/pipeline-complete.json is written (inf-portal-publisher has already
# fired off that sentinel) and BEFORE auto-shutdown. Because the sentinel is
# already persisted, an eclipse failure OR hang here cannot abort the chain,

# on A/B/C/D. Each step runs via run_eclipse_step (timeout-bounded + non-fatal).

    echo "  Portal publish already gated + fired on A/B/C/D. Eclipse outcome"
    echo "  below does NOT affect chain_exit ($chain_exit) or the publisher."
    echo "================================================================"

    for _ecl_id in $CHAIN_ECLIPSE; do
        run_eclipse_step "$_ecl_id"
    done
    ECLIPSE_RESULTS="${ECLIPSE_RESULTS# }"

    _eclipse_overall="ok"
    case "$ECLIPSE_RESULTS" in
        *:fail* | *:timeout* | *:missing*) _eclipse_overall="degraded" ;;
    esac
    echo ""

    _ecl_status_json=$(printf '{"overall":"%s","results":"%s","run_date":"%s","chain":"%s","chain_exit":%s,"completed_at":"%s"}' \

# ---------------------------------------------------------------------------
# Auto-shutdown: fires after all modes (chain/single/from), not dry-run.
# Watchdog-shutdown.sh handles unexpected exits; this handles normal
# completion. 2-minute grace period so S3 uploads can finish.
#
# Tag-aware branch: if this instance carries Tier=qa, skip Layer 1 shutdown.
# QA runners are manually managed -- auto-halt after a test run loses the
# instance state before the operator can inspect it.
#
# BRONZE-FIX #9 (the 21:25Z near-miss): the arm is now CONCURRENCY-AWARE.
# During the 2026-06-12 P0 recovery, a `--chain eclipse` dispatch completed
# and armed this shutdown WHILE the full recovery chain was still in flight
# on the same instance -- DS caught + cancelled it at 21:27Z, else the
# shutdown would have killed the live recovery. The guard below detects
# OTHER in-flight dispatch processes (by process GROUP -- each SSM
# RunCommand invocation runs in its own process group, while this dispatch's
# subshells/tee/telemetry children and the nohup'd --bg child share OURS)
# and DEFERS: the LAST dispatch to complete sees no others and arms.
#
# PRECISION MATTERS HERE: run-gold-layer.sh REMOVES the watchdog timer at
# boot (unit files deleted), so this arm is the instance's ONLY normal
# shutdown path -- a false "in-flight" reading (e.g. a transient --status
# poll counted as a chain) would defer the arm with NO successor and strand
# the instance running. The filter therefore excludes non-chain invocations
# (--status/--kill/--cancel-shutdown/--help/--dry-run) by cmdline. The
# converse (crashed dispatch never arms) is a PRE-EXISTING exposure of the
# watchdog removal, unchanged by this guard -- flagged as #9-F1.
# Alert-egress-before-arm (DS constraint 3): the arm/defer decision is
# logged + emitted as a CW metric BEFORE any shutdown command runs.
# ---------------------------------------------------------------------------

_other_dispatch_pids() {
    # PIDs of CHAIN-BEARING orion-gold-dispatch.sh processes OUTSIDE our
    # process group. pgrep/ps exist on Amazon Linux (EC2-side; Rule-16 N/A).
    _my_pgid=$(ps -o pgid= -p $$ 2>/dev/null | tr -d ' ')
    for _pid in $(pgrep -f '[o]rion-gold-dispatch\.sh' 2>/dev/null); do
        _pgid=$(ps -o pgid= -p "$_pid" 2>/dev/null | tr -d ' ')
        [ -z "$_pgid" ] || [ "$_pgid" = "$_my_pgid" ] && continue
        _args=$(ps -o args= -p "$_pid" 2>/dev/null)
        case "$_args" in
            *--status*|*--kill*|*--cancel-shutdown*|*--help*|*--dry-run*)
                continue ;;  # transient/non-chain invocation -- never defer on it
        esac
        printf '%s ' "$_pid"
    done
}

_emit_shutdown_metric() {
    # AC2 observability: Inflection/GoldRunner ShutdownArmed|ShutdownDeferred
    aws cloudwatch put-metric-data --namespace "Inflection/GoldRunner" \
        --metric-name "$1" --value 1 --unit Count \
        --dimensions Service=inf-gold-layer-runner >/dev/null 2>&1 || true
}
if [ "$DRY_RUN" = "0" ] && [ "$NO_SHUTDOWN" = "0" ]; then
    _IMDS_TOKEN=$(curl -s -m2 -X PUT "http://169.254.169.254/latest/api/token" \
        -H "X-aws-ec2-metadata-token-ttl-seconds: 60" 2>/dev/null || true)
    _INSTANCE_ID=$(curl -s -m2 \
        -H "X-aws-ec2-metadata-token: $_IMDS_TOKEN" \
        "http://169.254.169.254/latest/meta-data/instance-id" 2>/dev/null || true)
    if [ -n "$_INSTANCE_ID" ]; then
        _TIER_TAG=$(aws ec2 describe-tags \
            --filters "Name=resource-id,Values=$_INSTANCE_ID" "Name=key,Values=Tier" \
            --query 'Tags[0].Value' --output text 2>/dev/null || true)
        if [ "$_TIER_TAG" = "qa" ]; then
            echo "Tier=qa tag detected -- skipping Layer 1 self-shutdown (QA runner)."
            NO_SHUTDOWN=1
        fi
    fi
fi
if [ "$DRY_RUN" = "0" ] && [ "$NO_SHUTDOWN" = "0" ]; then
    # BRONZE-FIX #9 AC1+AC3: in-flight chain on this instance -> DO NOT arm.
    _INFLIGHT_PIDS=$(_other_dispatch_pids)
    if [ -n "$_INFLIGHT_PIDS" ]; then
        echo ""
        echo "[$(date -u +%Y-%m-%dT%H:%M:%SZ)] SHUTDOWN_DEFERRED chain=$CHAIN_LABEL other_dispatch_pids=${_INFLIGHT_PIDS% } -- NOT arming; the last chain-bearing dispatch to complete arms. The 21:25Z near-miss class is guarded."
        _emit_shutdown_metric ShutdownDeferred
    else
        echo ""
        # AC2: arm signal egresses (log + metric) BEFORE the shutdown command
        # (DS constraint 3 -- alert egress must not race the arm).
        echo "[$(date -u +%Y-%m-%dT%H:%M:%SZ)] SHUTDOWN_ARMED chain=$CHAIN_LABEL grace=2m -- cancel: 'orion-gold-dispatch.sh --cancel-shutdown' (idempotent) or 'sudo shutdown -c'."
        _emit_shutdown_metric ShutdownArmed
        sudo shutdown -h +2 2>/dev/null || true
    fi
fi

exit "$chain_exit"
