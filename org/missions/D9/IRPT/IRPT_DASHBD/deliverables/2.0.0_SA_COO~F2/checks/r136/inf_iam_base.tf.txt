# =============================================================================
# 1a. Scope account feed -- PutObject and GetObject on the two feed prefixes of
#     the assets bucket, ListBucket under an s3:prefix condition (IB-pr-603).
#     The grant is the signed packet's (NEVER_WIDEN_SIGNOFF_PACKET_SCOPE_FEED_IAM_PAIR,
#     deliverables/2.0.0_SA_COO~P2): this block changes the spelling, never the grant.
#     The policy name comes from the opal_name module (orion, inflection, scope-account-feed,
#     policy). Every source, scope, bucket and key prefix is read from infra/logic/registry.
#     The table name is the one value not read from the registry (it has no registry row).
#     The prefixes equal lib/opal_naming.py address() output for the bronze and silver layers.
# =============================================================================

module "scope_feed_policy_name" {
  source   = "../modules/opal_name"
  system   = "orion"
  scope    = "inflection"
  function = "scope-account-feed"
  kind     = "policy"
}

locals {
  scope_feed_systems = { for r in jsondecode(file("${path.module}/../../../logic/registry/systems.json")).systems : r.key => r }
  scope_feed_scopes  = { for r in jsondecode(file("${path.module}/../../../logic/registry/scopes.json")).scopes : r.key => r }
  scope_feed_tiers   = { for t, h in jsondecode(file("${path.module}/../../../logic/registry/tiers.json")).homes.FIRM : t => h }

  scope_feed_source = lookup(local.scope_feed_systems, "orion").key
  scope_feed_scope  = lookup(local.scope_feed_scopes, "inflection").long
  scope_feed_table  = "scope_account_returns"

  # the bucket of each layer: the FIRM tier home (tiers.json, doctrine R4: bronze and silver share one home)
  scope_feed_layers = {
    bronze = lookup(local.scope_feed_tiers, "bronze").bucket
    silver = lookup(local.scope_feed_tiers, "silver").bucket
  }
  scope_feed_keys = {
    for tier, bucket in local.scope_feed_layers : tier => "${local.scope_feed_source}/${local.scope_feed_scope}/${tier}/${local.scope_feed_table}"
  }
  scope_feed_object_arns = [
    for tier, bucket in local.scope_feed_layers : "arn:aws:s3:::${bucket}/${local.scope_feed_keys[tier]}/*"
  ]
  scope_feed_list_prefixes = [
    for tier, key in local.scope_feed_keys : "${key}/*"
  ]
  scope_feed_list_bucket_arns = distinct([
    for tier, bucket in local.scope_feed_layers : "arn:aws:s3:::${bucket}"
  ])
}

resource "aws_iam_role_policy" "inf_gold_layer_scope_feed" {
  name = module.scope_feed_policy_name.name
  role = aws_iam_role.inf_gold_layer.id

  policy = jsonencode({
    Version = "2012-10-17"
    Statement = [
      {
        Sid      = "S1PutAndReadScopeFeedKeysOnly"
        Effect   = "Allow"
        Action   = ["s3:PutObject", "s3:GetObject"]
        Resource = local.scope_feed_object_arns
      },
      {
        Sid      = "S2ListScopeFeedPrefixesOnly"
        Effect   = "Allow"
        Action   = ["s3:ListBucket"]
        Resource = local.scope_feed_list_bucket_arns
        Condition = {
          StringLike = { "s3:prefix" = local.scope_feed_list_prefixes }
        }
      }
    ]
  })
}
