// Scratch render for the record (not committed): mounts the page block in happy-dom and prints the markup it produces.
import { test } from 'bun:test'
import type { Ticket } from '../src/api/endpoints/tickets'

const REGISTRATOR = '@happy-dom/global-registrator'
const { GlobalRegistrator } = (await import(REGISTRATOR)) as { GlobalRegistrator: { register(): void; unregister(): Promise<void> } }
GlobalRegistrator.register()
const { createRoot } = await import('react-dom/client')
const { act, createElement } = await import('react')
const { AnalysisDraftAndPackage } = await import('../src/pages/TicketCenter/AnalysisRequestParts')
;(globalThis as { IS_REACT_ACT_ENVIRONMENT?: boolean }).IS_REACT_ACT_ENVIRONMENT = true

const WORDS = 'The report package could not be built, so nothing was attached. Press Retry analysis request to build it again.'
const base: Ticket = {
  ticket_id: 'TKT-0031', ticket_type: 'analysis_request', status: 'in_progress', title: 'Smith overview', description: '',
  requester: { email: '', name: '', firm: '' }, created_by: '', created_at: '', assignee: 'sam@opal', source: '',
  routing: { primary: [], secondary: [], visibility: [] }, events: [], answers: { firm: 'fir_test' },
  report_ref: { request_id: 'REQ-31', recorded_by: '', recorded_at: '' },
  draft_commentary: { text: 'Line one.', model: 'model-x', generated_at: '2026-10-09T01:02:03Z', status: 'DRAFT' },
}
const env = { publicReasonCode: 'PACKAGER_FAILED', safeActions: [{ action: 'none', label: WORDS, reasonCode: 'PACKAGER_FAILED' }] }

async function render(t: Ticket, label: string) {
  const container = document.createElement('div')
  document.body.appendChild(container)
  const root = createRoot(container)
  const ir = { getJobStatus: async () => env } as never
  await act(async () => { root.render(createElement(AnalysisDraftAndPackage, { ticket: t, ir })) })
  await act(async () => { await new Promise<void>(r => setTimeout(r, 25)) })
  console.log(`=== ${label} ===`)
  console.log(container.innerHTML)
  await act(async () => { root.unmount() })
  container.remove()
}

test('print markup', async () => {
  await render(base, 'no package, failure words')
  await render({ ...base, package: { report_ref: 'r', pdf_ref: 'p', original_ref: null, commentary_ref: 'c', attached_at: 'x', attached_by: 'y' } }, 'package attached')
  await GlobalRegistrator.unregister()
})
