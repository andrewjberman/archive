function getSentimentColor(sentiment: number): string {
  // Color definitions (RGB)
  const RED = { r: 245, g: 54, b: 92 }      // hsl(var(--danger)) - bearish
  // The neutral lean keeps its yellow family (Andrew-direct 2026-09-14 ~18:5xZ: the neutral, yellow-green
  // bullish lean and yellow-orange bearish lean "has been relegated to blue which is incorrect"). RB-s-4 had
  // moved these three stops onto the one-hue ramp; his word restores the pre-round values.
  const ORANGE = { r: 249, g: 115, b: 22 }  // yellow-orange - light bearish
  const YELLOW = { r: 234, g: 179, b: 8 }   // yellow - neutral
  const LIME = { r: 132, g: 204, b: 22 }    // yellow-green - light bullish
  const GREEN = { r: 45, g: 206, b: 137 }   // hsl(var(--success)) - bullish

  // Interpolate between two colors
  const lerp = (a: typeof RED, b: typeof RED, t: number) => ({
    r: Math.round(a.r + (b.r - a.r) * t),
    g: Math.round(a.g + (b.g - a.g) * t),
    b: Math.round(a.b + (b.b - a.b) * t),
  })

  const toHex = (c: typeof RED) =>
    `#${c.r.toString(16).padStart(2, '0')}${c.g.toString(16).padStart(2, '0')}${c.b.toString(16).padStart(2, '0')}`

  // Clamp sentiment to expected range
  const s = Math.max(-0.5, Math.min(0.5, sentiment))

  // Gradient mapping
  if (s >= 0.20) {
    return toHex(GREEN)  // Full green
  } else if (s >= 0.05) {
    // Yellow-green to green (0.05 to 0.20)
    const t = (s - 0.05) / 0.15
    return toHex(lerp(LIME, GREEN, t))
  } else if (s >= 0) {
    // Yellow to yellow-green (0 to 0.05)
    const t = s / 0.05
    return toHex(lerp(YELLOW, LIME, t))
  } else if (s >= -0.05) {
    // Light orange to yellow (-0.05 to 0)
    const t = (s + 0.05) / 0.05
    return toHex(lerp(ORANGE, YELLOW, t))
  } else if (s >= -0.30) {
    // Red to light orange (-0.30 to -0.05)
    const t = (s + 0.30) / 0.25
    return toHex(lerp(RED, ORANGE, t))
  } else {
    return toHex(RED)  // Full red
  }
}
