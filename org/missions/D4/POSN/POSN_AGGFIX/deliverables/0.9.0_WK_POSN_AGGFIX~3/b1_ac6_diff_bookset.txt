+/** True when the data carries an aggregate block with at least one holding. */

+/**
+ * Merge the Opal-only aggregate into the models payload the page renders.
+ *
+ * ONE rule, from the 1.1 contract: the aggregate is offered ONLY when the file says
+ * `status: "OK"` AND `current_holdings.AGG.holdings` is non-empty. Anything else --
+ * UNAVAILABLE, an empty block, or no payload because the query never ran for this viewer --
+ * returns the input object ITSELF, so a viewer who is not offered the aggregate renders from
+ * exactly the object they rendered from before.
+ *
+ * Pure: the input is never mutated. A merge returns a new object whose new
+ * `current_holdings` carries the five books' entries by reference and the aggregate block.
+ */
+export const mergeOpalAggregate = (
+  data: PositioningDataResponse | null,
+  payload: OpalAggregateResponse | null | undefined
+): PositioningDataResponse | null => {
+  if (!data || !payload || payload.status !== 'OK') return data
+  const block = payload.current_holdings?.[AGGREGATE_BOOK]
+  if (!block || !Array.isArray(block.holdings) || block.holdings.length === 0) return data
+  return {
+    ...data,
+    current_holdings: { ...(data.current_holdings ?? {}), [AGGREGATE_BOOK]: block },
+  }
+}
+
+/**
+ * The book set FOR THIS VIEWER: `resolveBooks` unchanged, with the aggregate FIRST when, and
+ * only when, all three hold -- the family is one the aggregate rolls up, the viewer passes
+ * the page's internal predicate, and the data carries an aggregate with holdings. For every
+ * other viewer and state the result IS `resolveBooks`, so the five books cannot move.
+ */
+export const resolveViewerBooks = (
+  mode: PositioningMode,
+  availableCodes: readonly string[] | null | undefined,
+  viewer: { internal: boolean },
+  data: PositioningDataResponse | null | undefined
+): string[] => {
+  const books = resolveBooks(mode, availableCodes)
+  if (!(AGGREGATE_ROLLS_UP[mode] && viewer.internal && hasOpalAggregate(data))) return books

+}

+/**

+export const rememberedBookFor = (

+/**
+ * Where a viewer lands: the remembered book when this viewer's set holds it and it is not a
+ * book-level shell; else the ordinary default -- PREFERRED_DEFAULT_BOOK where the set has it
+ * and it carries rows, else the first book that carries rows, else the first book. These are
+ * the layers HoldingsTab's `defaultBook` applies today, stated as a pure function so the
+ * remembered-choice check can be exercised without a browser.
+ */
+export const landingBook = (
+  remembered: string | null | undefined,
+  books: readonly string[],
+  bookStatus: Record<string, 'full' | 'book_only'> = {}
+): string => {
+  const carriesRows = (b: string) => bookStatus[b] !== 'book_only'
+  const kept = rememberedBookFor(remembered, books)
+  if (kept && carriesRows(kept)) return kept
+  if (books.includes(PREFERRED_DEFAULT_BOOK) && carriesRows(PREFERRED_DEFAULT_BOOK)) return PREFERRED_DEFAULT_BOOK
+  return books.find(carriesRows) ?? books[0] ?? ''
+}
