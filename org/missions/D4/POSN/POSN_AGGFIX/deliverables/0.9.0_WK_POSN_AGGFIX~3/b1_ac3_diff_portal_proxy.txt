+// ============================================================================
+// Opal aggregate (POSN_AGGFIX, cell AF1-s-12) -- OPAL VIEWERS ONLY

+//
+// A SEPARATE served key from positioning_data.json. The aggregate is Opal's own firm-wide

+// getPositioningData and the shared file are untouched by this section.
+//
+// Contract 1.1 (POSN_AGGFIX decomposition s.1.1). The page offers an aggregate ONLY when
+// `status === 'OK'` and `current_holdings.AGG.holdings` is non-empty
+// (pages/Positioning/shared/bookSet.ts mergeOpalAggregate).
+
+export type OpalAggregateStatus = 'OK' | 'UNAVAILABLE'
+
+/** A value per primary model, keyed by the five canonical model codes. */
+export interface OpalAggregateByModel {
+  UST: number
+  ST: number
+  INT: number
+  LT: number
+  ULT: number
+}
+
+export interface OpalAggregateBlend {
+  model_name: string
+  aum: number
+  constituents: Record<string, number>
+}
+
+export interface OpalAggregateResponse {
+  status: OpalAggregateStatus
+  /** null when OK; one line saying why when UNAVAILABLE. */
+  reason: string | null
+  generated_at: string
+  data_as_of: string
+  /** Exactly today's AGG holding shape: CurrentPortfolioHoldings, each holding carrying model_breakdown. */
+  current_holdings: { AGG: CurrentPortfolioHoldings }
+  weights: OpalAggregateByModel
+  weights_source: { key: string; version_id: string; period: string }
+  members: OpalAggregateByModel & { blends: OpalAggregateBlend[] }
+  left_out: Array<{ model_name: string; aum: number }>
+  reconciliation: {
+    five_primary_sum: number
+    members_sum: number
+    file_total_model_aum: number
+    gap: number
+    gap_explanation_ref: string
+  }
+}
