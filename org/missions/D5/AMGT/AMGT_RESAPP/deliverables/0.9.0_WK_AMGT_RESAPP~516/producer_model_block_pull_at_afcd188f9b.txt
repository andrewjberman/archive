def _bulk_text_value(val: Any) -> Any:
    """Join a BULK field's row-dict list into ONE string (rows in order, single spaces).
    A non-list value passes through unchanged, so this is safe to call unconditionally on
    any header value, not only ones known in advance to be bulk-shaped."""
    if not isinstance(val, list):
        return val
    parts: list[str] = []
    for row in val:
        if isinstance(row, dict):
            for v in row.values():
                if v not in (None, ""):
                    parts.append(str(v))
        elif row not in (None, ""):
            parts.append(str(row))
    return " ".join(parts) if parts else None

def _iso_date(val: Any) -> str | None:
    """A value as an ISO YYYY-MM-DD date, or None when it is not unambiguously a date.
    Accepts a date/datetime object, an ISO string (with or without a time part), YYYYMMDD and YYYY/MM/DD."""
    if isinstance(val, _dt.datetime):
        return val.date().isoformat()
    if isinstance(val, _dt.date):
        return val.isoformat()
    if not isinstance(val, str):
        return None
    text = val.strip()
    match = re.fullmatch(r"(\d{4})(\d{2})(\d{2})", text) or re.fullmatch(
        r"(\d{4})([-/])(\d{2})\2(\d{2})(?:[T ].*)?", text)
    if not match:
        return None
    parts = [g for g in match.groups() if g not in ("-", "/")]
    try:
        return _dt.date(int(parts[0]), int(parts[1]), int(parts[2])).isoformat()
    except ValueError:
        return None


def _ern_announcements_value(val: Any) -> tuple[list[dict] | None, int]:
    """Turn the bulk earnings field's row-dict list into [{"announcement_date", "period"}...], in
    Bloomberg's row order. Per row: the DATE column is the first key whose value reads as a date (a key
    named like 'date' is preferred); the PERIOD is the first other non-empty value, as Bloomberg's text.
    A row whose date cannot be read is left out and counted. Returns (rows or None, left_out)."""
    if not isinstance(val, list) or not val:
        return None, 0
    out: list[dict] = []
    left_out = 0
    for row in val:
        if not isinstance(row, dict):
            left_out += 1
            continue
        keys = sorted(row, key=lambda k: 0 if "date" in str(k).lower() else 1)  # stable: row order within a class
        date_key = next((k for k in keys if _iso_date(row[k]) is not None), None)
        if date_key is None:
            left_out += 1
            continue
        period = next((row[k] for k in row if k != date_key and row[k] not in (None, "")), None)
        out.append({"announcement_date": _iso_date(row[date_key]), "period": None if period is None else str(period)})
    return (out or None), left_out

def security_key(security: str) -> str:
    return security.replace(" ", "_")


def utc_stamp() -> str:
    return _dt.datetime.utcnow().strftime("%Y%m%dT%H%M%SZ")


def _rolling_end_date() -> str:
    """AMGT-rk-35: the request's end date, computed fresh at call time -- today's UTC date,
    never a fixed constant (C-MODEL-WINDOW-AND-ROWS clause (1))."""
    return _dt.datetime.utcnow().strftime("%Y-%m-%d")


def _s3():
    return boto3.client("s3")


def s3_put_if_absent(s3, key: str, body: bytes, content_type: str = "application/json") -> None:
    try:
        s3.head_object(Bucket=BUCKET, Key=key)
        exists = True
    except botocore.exceptions.ClientError as e:
        code = e.response.get("Error", {}).get("Code", "")
        if code in ("404", "NoSuchKey", "NotFound"):
            exists = False
        else:
            raise
    if exists:
        raise RuntimeError(
            f"REFUSING TO OVERWRITE existing key (STORE_CONTRACT rule 1): s3://{BUCKET}/{key}"
        )
    s3.put_object(Bucket=BUCKET, Key=key, Body=body, ContentType=content_type)


def _read_json_from_s3(s3, key: str) -> Any:
    obj = s3.get_object(Bucket=BUCKET, Key=key)
    return json.loads(obj["Body"].read())


class NoRawStampError(RuntimeError):
    """AMGT-rk-70: raised by `_latest_stamp` when bronze objects exist for a security but none
    of the stamps carries both a `refdata` and a `histdata` bronze object -- an object existing
    for one request kind is not evidence the pair landed (row AMGT_RESAPP-Z-W38-S5, measured
    2026-09-19). A RuntimeError subclass so every existing `except RuntimeError` still catches
    it."""


def _bronze_prefix(request: str, sec_key: str) -> str:
    """AMGT-rk-90 (M1): the builder's bronze prefix for one request kind and security --
    `bloomberg/harvest/bronze/<request>/security_key=<safe>/` -- via `hstore.table_prefix` /
    `hstore.entity_partition` only, never a bucket or key literal."""
    return hstore.table_prefix(VENDOR, "bronze", request) + hstore.entity_partition("security_key", sec_key) + "/"


def _bronze_key(request: str, sec_key: str, stamp: str) -> str:
    """The builder's bronze key for one request/security/stamp -- `<prefix><stamp>.json`."""
    return f"{_bronze_prefix(request, sec_key)}{stamp}.json"


def _state_pull_log_prefix(sec_key: str) -> str:
    """The builder's state prefix for pull logs -- `amgt/harvest/state/pull_log/security_key=<safe>/`."""
    return hstore.table_prefix("amgt", "state", "pull_log") + hstore.entity_partition("security_key", sec_key) + "/"


def _bronze_stamp_set(s3, request: str, sec_key: str) -> set[str]:
    """Every stamp (object name minus `.json`) landed for this request/security under the new
    bronze grammar -- a flat listing, never a folder-of-files walk (the old grammar's shape)."""
    prefix = _bronze_prefix(request, sec_key)
    resp = s3.list_objects_v2(Bucket=BUCKET, Prefix=prefix)
    return {
        key[len(prefix):-len(".json")]
        for key in (o["Key"] for o in resp.get("Contents", []) or [])
        if key.endswith(".json")
    }


def _stamps_with_raw_pair(s3, sec_key: str) -> list[str]:
    """Every stamp, newest first, that ACTUALLY carries both a `refdata` and a `histdata` bronze
    object for `sec_key` (never inferred from one side alone). Mirrors
    `build_unit_map._stamps_with_refdata` EXACTLY (see that function's docstring for the measured
    hazard); duplicated here rather than imported because `build_unit_map.py` already does
    `import model_block_pull as _mbp`, so the reverse import would be circular.
    `test_mirror_agrees_with_build_unit_map` is the standing regression that the two copies do
    not drift (round 16's AMGT-rk-65 exists because exactly that happened to the accession
    rule)."""
    ref_stamps = _bronze_stamp_set(s3, "refdata", sec_key)
    hist_stamps = _bronze_stamp_set(s3, "histdata", sec_key)
    return sorted(ref_stamps & hist_stamps, reverse=True)


def _latest_stamp(s3, sec_key: str) -> str:
    any_stamps: set[str] = set()
    for request in hbloom.BRONZE_REQUESTS:
        any_stamps |= _bronze_stamp_set(s3, request, sec_key)
    if not any_stamps:
        raise RuntimeError(f"no bronze stamp found for {sec_key!r} under s3://{BUCKET}/{_NEW_BRONZE_ROOT}")
    qualifying = _stamps_with_raw_pair(s3, sec_key)
    if not qualifying:
        raise NoRawStampError(
            f"no raw stamp qualifies for {sec_key!r} under s3://{BUCKET}/{_NEW_BRONZE_ROOT}: "
            f"{len(any_stamps)} stamp(s) rejected (no refdata+histdata pair)"
        )
    return qualifying[0]


def _end_requested_from_pull_log(s3, sec_key: str, raw_stamp: str) -> str | None:
    """AMGT-rk-35 item 2(iii): `history_window.end_requested` is READ FROM THE RAW REQUEST
    RECORD this producer's OWN pull already wrote (pull_log, `:stage_pull`), never recomputed at
    derive time -- so a derive re-run against an older raw stamp reports the window that stamp
    was ACTUALLY pulled under, not today's date. Reads STATE (AMGT-rk-90 M1: pull_log moved off
    the old vendor-partitioned raw prefix)."""
    key = f"{_state_pull_log_prefix(sec_key)}{raw_stamp}.json"
    try:
        pull_log = _read_json_from_s3(s3, key)
    except botocore.exceptions.ClientError:
        return None
    for req in pull_log.get("requests", []):
        if req.get("type") == "HistoricalDataRequest":
            return req.get("end_date")
    return None


def _hist_field_chunks(fields: list[str], per_request: int) -> list[list[str]]:
    """CRPS-f-2: split `fields` into chunks of at most `per_request`, field order preserved
    (the first `per_request` fields, then the rest) -- never re-sorted, never de-duplicated."""
    return [fields[i:i + per_request] for i in range(0, len(fields), per_request)]


def _merge_historical_by_date(chunk_results: list[dict[str, list[dict]]]) -> dict[str, list[dict]]:
    """CRPS-f-2: merge N `historical_data()` chunk results (one per HIST_FIELDS_PER_REQUEST-sized
    field slice, same securities/window/periodicity) into ONE result of today's shape
    `{security: [{"date": ..., field: value, ...}]}`. A date seen in any chunk appears once,
    carrying every field any chunk returned for it (later chunks never overwrite an earlier
    chunk's value for a field it already supplied, since chunks never share a field)."""
    merged: dict[str, dict[str, dict]] = {}
    for chunk in chunk_results:
        for sec, rows in chunk.items():
            by_date = merged.setdefault(sec, {})
            for row in rows:
                date = row.get("date")
                rec = by_date.setdefault(date, {"date": date})
                for k, v in row.items():
                    if k == "date":
                        continue
                    rec.setdefault(k, v)
    return {sec: [by_date[d] for d in sorted(by_date.keys())] for sec, by_date in merged.items()}


def stage_pull(blp, s3, security: str) -> dict:
    """Fetch header + fiscal-year history from Bloomberg DAPI; land the raw response bytes.

    AMGT-rk-90 (M1): lands through the harvest grammar -- `hbloom.bronze_write` for
    refdata/histdata/bestdata (body `hbloom.client_json(...)`, the ONE deterministic
    serialisation of what the Terminal client handed this producer, C-BRONZE-AS-RECEIVED) and
    `hbloom.state_put_pull_log` for the pull log (STATE, never bronze -- `bronze_write` itself
    refuses `request="pull_log"`). The pull log is the SAME class as the raw landing it records
    (C-PRODUCTION-HOME): ONE object per pull, beside that landing -- never a cumulative repo
    file (the input ledger already records every pull, :246).

    AMGT-rk-36 adds THREE ReferenceDataRequests, one per fiscal period, each with an EXPLICIT
    BEST_FPERIOD_OVERRIDE -- never sent bare (item 1: "a request without it is a violation, not
    a shortcut"). All three land UNTOUCHED as ONE object, bestdata.json, in the SAME stamp."""
    sec_key = security_key(security)
    stamp = utc_stamp()
    end_date = _rolling_end_date()

    pull_log = {"securities": [security], "requests": []}
    ref = blp.reference_data([security], HEADER_FIELDS)
    pull_log["requests"].append(
        {"type": "ReferenceDataRequest", "security": security, "fields": HEADER_FIELDS, "at": stamp}
    )
    # CRPS-f-2: HIST_FIELDS (39) is split into chunks of at most HIST_FIELDS_PER_REQUEST (25)
    # fields, one HistoricalDataRequest per chunk, same security/window/periodicity -- a live
    # pull on 2026-09-25 returned {} history for every security when all 39 rode one request.
    # `on_response_error="raise"` so a rejected request stops this security's pull loudly
    # instead of landing `{}` (blp_client.py's handle() used to drop a request-level error).
    hist_field_chunks = _hist_field_chunks(HIST_FIELDS, HIST_FIELDS_PER_REQUEST)
    n_chunks = len(hist_field_chunks)
    hist_chunk_results = []
    for chunk_i, chunk_fields in enumerate(hist_field_chunks, start=1):
        chunk_result = blp.historical_data(
            [security], chunk_fields, HIST_START, end_date, periodicity=PERIODICITY,
            on_response_error="raise",
        )
        hist_chunk_results.append(chunk_result)
        pull_log["requests"].append(
            {
                "type": "HistoricalDataRequest",
                "security": security,
                "fields": chunk_fields,
                "chunk": chunk_i,
                "chunks_total": n_chunks,
                "start_date": HIST_START,
                "end_date": end_date,
                "periodicity": PERIODICITY,
                "at": stamp,
            }
        )
    hist = _merge_historical_by_date(hist_chunk_results)

    best_fields = FORECAST_FIELDS + [FORECAST_DATE_FIELD]
    best_doc: dict = {"requests": []}
    for fp in FORECAST_OVERRIDES:
        overrides = {FORECAST_OVERRIDE_FIELD: fp}
        resp = blp.reference_data([security], best_fields, overrides=overrides)
        best_doc[fp] = resp
        req_record = {
            "type": "ReferenceDataRequest",
            "security": security,
            "fields": best_fields,
            "overrides": overrides,
            "at": stamp,
        }
        best_doc["requests"].append(req_record)
        pull_log["requests"].append(req_record)

    fetched_at = _dt.datetime.strptime(stamp, "%Y%m%dT%H%M%SZ").replace(tzinfo=_dt.timezone.utc)
    ref_receipt = hbloom.bronze_write("refdata", sec_key, fetched_at, hbloom.client_json(ref), client=s3)
    hist_receipt = hbloom.bronze_write("histdata", sec_key, fetched_at, hbloom.client_json(hist), client=s3)
    best_receipt = hbloom.bronze_write("bestdata", sec_key, fetched_at, hbloom.client_json(best_doc), client=s3)
    pull_log_receipt = hbloom.state_put_pull_log(
        sec_key, stamp, json.dumps(pull_log, default=str, indent=2).encode("utf-8"), client=s3
    )
    ref_key, hist_key, best_key, pull_log_key = (
        ref_receipt["key"], hist_receipt["key"], best_receipt["key"], pull_log_receipt["key"]
    )
    _load_ledger().record_run(sec_key, "market_data", VENDOR, "OK", stamp, f"pull landed at {ref_key}, {hist_key}, {best_key}")
    return {

def merge_by_fiscal_year(hist_rows: list[dict]) -> dict[str, dict]:
    """C-FISCAL-YEAR-JOIN: group historical rows by the calendar year of `date`. Price and
    fundamentals can land on different dates within the same fiscal year (F5) -- grouping by
    year (rather than the exact date string) merges those sub-rows into one fiscal-year record
    instead of leaving a price-only row and a fundamentals-only row unjoined."""
    groups: dict[str, dict] = {}
    for row in hist_rows:
        date = row.get("date")
        if not date:
            continue
        fy = date[:4]
        g = groups.setdefault(fy, {"fiscal_year": fy, "_dates": []})
        g["_dates"].append(date)
        for k, v in row.items():
            if k == "date":
                continue
            g[k] = v
    return groups


def _sign_label(value) -> str:
    if isinstance(value, (int, float)):
        if value > 0:
            return "POSITIVE"
        if value < 0:
            return "NEGATIVE"
        return "ZERO"
    return "N/A (non-numeric field)"

def _scalar_exception(entry: dict, security: str, axis: str) -> str | None:
    """CRPS-u-1: a SCALAR unit/sign is the field's shared value, but per_security_detail[security] can
    withdraw it for that one security -- an `exception` carrying a `unit`/`sign` reason (build_unit_map.py),
    or (older entries) a `<axis>_status` that is not MEASURED. Returns the UNRESOLVED-ON text, else None."""
    detail = (entry.get("per_security_detail") or {}).get(security)
    if not isinstance(detail, dict):
        return None
    exc = detail.get("exception")
    reason = exc.get(axis) if isinstance(exc, dict) else None
    if reason:
        reason = str(reason)
        if reason.startswith("UNRESOLVED"):
            return reason
        return "UNRESOLVED-ON-%s: %s" % (security, reason)
    status = detail.get(axis + "_status")
    if status and status != "MEASURED":
        status = str(status)
        if status.startswith("UNRESOLVED"):
            return status
        return "UNRESOLVED-ON-%s: %s" % (security, status)
    return None


def _resolved_unit(mnemonic: str, security: str) -> str:
    """The per-field, per-security MEASURED unit from field_map.json (build_field_map.py --
    C-UNIT-MAP-MEASURED), read back rather than re-derived, so the two artifacts never disagree
    silently. Falls back to a pointer if field_map.json has not been built yet (re-runnable
    standalone: `derive` does not require `build_field_map.py` to have run first)."""
    fm = _field_map()
    if fm:
        for entry in fm.get("fields", []):
            if entry.get("mnemonic") == mnemonic:
                unit = entry.get("unit")
                if isinstance(unit, dict):
                    u = unit.get(security)
                    if u:
                        return u
                elif unit:
                    exc = _scalar_exception(entry, security, "unit")
                    return exc if exc else unit
    return "SEE field_map.json (per-field, per-security MEASURED unit -- not yet built)"


def _resolved_sign(mnemonic: str, security: str, observed_value) -> str:
    """CRPS-y-5: the sign of THIS figure. (a) a map sign that is UNRESOLVED (the entry's own status or a
    per-security exception) is kept unchanged; (b) else build_unit_map.classify_sign on THIS value, MEASURED
    label returned (carries the convention words: NET_DEBT, CAPITAL_EXPEND); (c) else _sign_label. The
    map's sampled sign is never served on a figure whose own value contradicts it."""
    fm = _field_map()
    if fm:
        for entry in fm.get("fields", []):
            if entry.get("mnemonic") == mnemonic:
                sign = entry.get("sign")
                s = None
                if isinstance(sign, dict):
                    s = sign.get(security)
                elif sign:
                    s = _scalar_exception(entry, security, "sign") or sign
                if s and str(s).startswith("UNRESOLVED"):
                    # CRPS-ad-2: a map sign UNRESOLVED ONLY because the one sampled value was exactly zero
                    # says nothing about THIS figure -- the sample was a zero year, the sign convention of
                    # the series is not in doubt (BA's DPS: four positive years, then zeros). Such a figure
                    # is signed by its own value below (rules b, c), exactly as a resolved-map figure is.
                    # Every other UNRESOLVED reason (unit unresolved, absent value, ...) is kept unchanged.
                    try:
                        zero_reason = _load_build_unit_map().ZERO_SAMPLE_REASON
                    except Exception:
                        zero_reason = None
                    if not (zero_reason and zero_reason in str(s)):
                        return s
    try:
        label, status, _reason = _load_build_unit_map().classify_sign(mnemonic, observed_value, security)
        if status == "MEASURED" and label:
            return label
    except Exception:
        pass
    return _sign_label(observed_value)

def _currency_caveat(mnemonic: str, security: str) -> str | None:
    """If build_field_map.py's self-consistent PE cross-check found this SECURITY's

def _figure_currency(mnemonic: str, quote_ccy: str | None, reporting_ccy: str | None,
                     security: str | None = None) -> tuple[str | None, str]:
    """RULED item 2: classify ONE figure's currency and currency_basis.
      - a `_CURRENCY_SENSITIVE_FIELDS` member (a statement/per-share money figure) is stated in
        the REPORTING currency;
      - `PX_LAST` is stated in the QUOTE currency;
      - everything else (a ratio, a percent, a share count, a text/classification field, and
        the two currency-code header fields themselves) is currency-invariant.
    An UNSTATED needed header field is NEVER silently filled in -- the figure's `currency`
    stays None and `currency_basis` names why.

    CRPS-aa-1 (round 25): with `security` (Bloomberg form, e.g. "TSM US Equity") the three
    census figures follow the MEASURED denomination -- a `quote` class is stated in the quote
    currency (basis "quote (CRNCY)"), a figure the census left UNRESOLVED prints NO currency and
    a basis beginning "UNMEASURED:". Every other figure, and every call without `security`, is
    exactly the rule above.

    CRPS-am-1 (round 37): every MONEY figure now carries a currency too -- the members of
    `_MONEY_REPORTING_FIELDS` the reporting currency, `_MONEY_QUOTE_FIELDS` the quote currency,
    CRNCY_ADJ_MKT_CAP exactly as CUR_MKT_CAP. A needed header currency that is absent keeps the
    `UNSTATED:` null. Only ratios, percents, share counts, text, dates and years stay invariant."""
    if mnemonic in _MONEY_FOLLOWS_CUR_MKT_CAP:
        mnemonic = "CUR_MKT_CAP"
    # CRPS-an-1 (round 38, row Z-W1-S35): a count (shares) or ratio/percent carries NO currency.
    if mnemonic in _COUNT_RATIO_NO_CURRENCY_FIELDS:
        return None, _CURRENCY_BASIS_INVARIANT
    if security is not None:
        held = _DENOMINATION_UNRESOLVED.get(mnemonic, {}).get(security)
        if held is not None:
            return None, "UNMEASURED: %s (CRPS-aa-1 round-25 census)" % _DENOMINATION_REASONS[held]
        if security in _DENOMINATION_QUOTE.get(mnemonic, ()):
            if quote_ccy is None:
                return None, "UNSTATED: quote currency (CRNCY) not available"
            return quote_ccy, _CURRENCY_BASIS_QUOTE
    if (mnemonic in _CURRENCY_SENSITIVE_FIELDS or mnemonic in _DENOMINATION_REPORTING_MNEMONICS
            or mnemonic in _MONEY_REPORTING_FIELDS):
        if reporting_ccy is None:
            return None, "UNSTATED: reporting currency (EQY_FUND_CRNCY) not available"
        return reporting_ccy, _CURRENCY_BASIS_REPORTING
    if mnemonic == "PX_LAST" or mnemonic in _MONEY_QUOTE_FIELDS:
        if quote_ccy is None:
            return None, "UNSTATED: quote currency (CRNCY) not available"
        return quote_ccy, _CURRENCY_BASIS_QUOTE
    return None, _CURRENCY_BASIS_INVARIANT


def _ensure_unit_map(s3, security: str, sec_key: str, header_vals: dict, hist_rows: list, raw_stamp: str) -> None:
    """AMGT-rk-35 item 6 -- units at instantiation. When the field map carries NO entry yet for
    `security`, build one THROUGH AMGT-rk-34's builder (build_unit_map.py), merging the result
    into the map's committed home beside this producer. A security already present is left
    alone here -- the builder's own additive-merge rule (never overwrite a MEASURED entry) is
    what governs a re-measure, and this producer never re-measures on its own.

    AMGT-rk-36 item 4: the builder's `sample_value` reads a mnemonic from the HEADER dict
    first. The five BEST_ mnemonics live in a THIRD raw file (bestdata.json) this builder never
    reads on its own, so a representative sample (the FY1-override row, when it exists) is
    merged into a COPY of header_vals before the builder runs -- never the real ref object, and
    never written back to it. A missing/absent bestdata.json (every covered name predating this
    cell) leaves the merge a no-op and the builder correctly reports UNRESOLVED for those five."""
    fm = _field_map() or {}
    if security in set(fm.get("securities") or []):
        return
    bum = _load_build_unit_map()
    # CRPS-u-2: the CIE_DES_BULK join and the bestdata FY1 merge live in ONE function the
    # builder's own CLI also uses, so the two entry points build the same map (row Z-W2-S14).
    header_for_units = bum.prepare_header(s3, security, sec_key, header_vals, raw_stamp)
    resolved = {sec_key: (security, header_for_units, hist_rows, raw_stamp)}
    doc, summaries = bum.build_entries(resolved, remeasure=False)
    # newline="\n": Path.write_text's default text mode CRLF-ifies on this device -- a repo
    # JSON file must stay LF (CLAUDE.md; the memory bank's own "Python text-mode CRLF-ifies
    # repo files" entry).
    with open(bum.FIELD_MAP_PATH, "w", encoding="utf-8", newline="\n") as _fh:
        _fh.write(json.dumps(doc, indent=2, default=str))
    global _FIELD_MAP_CACHE
    _FIELD_MAP_CACHE = None  # invalidate so _resolved_unit/_resolved_sign re-read the fresh file
    for sk, sec, n_new in summaries:
        print(f"UNIT MAP: {sec} ({sk}) -- {n_new} field entr{'y' if n_new == 1 else 'ies'} built by build_unit_map.py")


def _crosswalk_sa_symbol(s3, security: str) -> str | None:
    """R14-P1's own field, READ, NEVER REQUIRED (item 5(a)). Looks at the newest
    crosswalk/v1/*.json for a record naming this security and returns its `sa_symbol` if
    landed; None on any miss (no crosswalk object yet, no matching record, or the field has
    not landed) -- an optional lookup, never a hard dependency of this producer."""
    try:
        resp = s3.list_objects_v2(Bucket=BUCKET, Prefix="crosswalk/v1/")
        keys = sorted(o["Key"] for o in resp.get("Contents", []))
        if not keys:
            return None
        doc = _read_json_from_s3(s3, keys[-1])
        sec_key = security_key(security)
        for rec in doc.get("records") or []:
            if rec.get("bloomberg") == security or rec.get("security_key") == sec_key:
                return rec.get("sa_symbol")
    except Exception:  # noqa: BLE001 -- an optional lookup never fails this producer's own act
        return None
    return None


def _forecast_period_end_dates(best: dict, security: str) -> dict:
    """{"FY1": date_or_None, "FY2": ..., "FY3": ...} read from the bestdata.json doc."""
    return {
        FORECAST_OVERRIDE_TO_PERIOD[ov]: ((best.get(ov) or {}).get(security) or {}).get(FORECAST_DATE_FIELD)
        for ov in FORECAST_OVERRIDES
    }


def _forecast_annual_spacing_ok(period_end_dates: dict, latest_closed_fy: str | None) -> bool:
    """The quarter-trap guard (item 2): every PRESENT period end date must fall in a year
    later than the block's latest_closed_fiscal_year, and every pair of PRESENT dates (in
    FY1/FY2/FY3 order) must be at least FORECAST_MIN_ANNUAL_GAP_DAYS apart. Fewer than two
    present dates cannot discriminate spacing on their own and this returns True -- the
    per-cell 'no BEST_PERIOD_END_DATE' rule already accounts for a missing date; an
    unparseable date is treated as a failure (never silently ignored)."""
    dates = []
    for label in ("FY1", "FY2", "FY3"):
        raw = period_end_dates.get(label)
        if not raw:
            continue
        try:
            dates.append(_dt.date.fromisoformat(str(raw)[:10]))
        except ValueError:
            return False
    if latest_closed_fy:
        try:
            latest_year = int(latest_closed_fy)
            if any(d.year <= latest_year for d in dates):
                return False
        except ValueError:
            pass
    for i in range(len(dates) - 1):
        if (dates[i + 1] - dates[i]).days < FORECAST_MIN_ANNUAL_GAP_DAYS:
            return False
    return True


def _fiscal_year_from_period_end(period_end_date: str) -> int | None:
    """Bloomberg names a fiscal year by its END year (item 2): a March 2027 period end is
    FY2027."""
    try:
        return _dt.date.fromisoformat(str(period_end_date)[:10]).year
    except (ValueError, TypeError):
        return None


def _derive_forecast_figures(
    s3,
    security: str,
    sec_key: str,
    raw_stamp: str,
    header_vals: dict,
    latest_closed_fy: str | None,
    vintage: str,
    sa_fallback: bool = True,
) -> tuple[list[dict], list[dict], dict]:
    """Section 1 items 2 and 5: the forecast-scope figures, their unresolved cells, and the
    forecast_fallback record. Reads bestdata.json BACK from S3 for raw_stamp (never an
    in-memory pull result), matching the re-runnability rule of the rest of this stage."""
    # AMGT-rk-112, C-CURRENCY-LABEL: same header dict, same classifier as stage_derive's own
    # header/fiscal-year figures (:1044-1045) -- no BEST_ mnemonic is a _CURRENCY_SENSITIVE_FIELDS
    # member, so every forecast figure classified invariant under _figure_currency as ruled
    # item 2 stated it. CRPS-am-1 (round 37): the BEST_* money mnemonics are now members of
    # _MONEY_REPORTING_FIELDS and carry the reporting currency.
    quote_ccy, reporting_ccy = _header_currencies(header_vals)
    # CRPS-y-2: with sa_fallback=False the fallback module is never loaded and no SA request is
    # made; a gap stays a labelled gap and the record says HELD_OFF.
    fb = _load_sa_forecast_fallback() if sa_fallback else None

    def _no_gap_record() -> dict:
        if fb is not None:
            return fb.no_gap_record()
        return {
            "sa_symbol_candidates": [],
            "sa_symbol": None,
            "entity_test": None,
            "classification": None,
            "gaps_before": 0,
            "filled_by_sa": 0,
            "sa_raw_key": None,
            "reason": "no Bloomberg gap -- nothing to fill",
            "sa_field_map": {},
        }

    figures: list[dict] = []
    unresolved: list[dict] = []
    best_key = _bronze_key("bestdata", sec_key, raw_stamp)
    try:
        best = _read_json_from_s3(s3, best_key)
    except botocore.exceptions.ClientError:
        unresolved.append(
            {
                "scope": "forecast",
                "fiscal_period": None,
                "mnemonic": None,
                "reason": f"raw stamp {raw_stamp} carries no bestdata.json (pulled before C-MODEL-FORECASTS)",
                "fallback_status": None,
            }
        )
        return figures, unresolved, _no_gap_record()

    period_end_dates = _forecast_period_end_dates(best, security)
    if not _forecast_annual_spacing_ok(period_end_dates, latest_closed_fy):
        unresolved.append(
            {
                "scope": "forecast",
                "fiscal_period": None,
                "mnemonic": None,
                "reason": FORECAST_NOT_ANNUAL_REASON,
                "fallback_status": None,
            }
        )
        return figures, unresolved, _no_gap_record()

    gaps: list[dict] = []
    for ov in FORECAST_OVERRIDES:
        label = FORECAST_OVERRIDE_TO_PERIOD[ov]
        row = (best.get(ov) or {}).get(security) or {}
        ped = row.get(FORECAST_DATE_FIELD)
        if not ped:
            for mnemonic in FORECAST_FIELDS:
                unresolved.append(
                    {
                        "scope": "forecast",
                        "fiscal_period": label,
                        "mnemonic": mnemonic,
                        "reason": "no BEST_PERIOD_END_DATE for this period",
                        "fallback_status": "ABSENT: no period end date",
                    }
                )
            continue
        fiscal_year = _fiscal_year_from_period_end(ped)
        for mnemonic in FORECAST_FIELDS:
            val = row.get(mnemonic)
            if val not in (None, ""):
                f_cur, f_cur_basis = _figure_currency(mnemonic, quote_ccy, reporting_ccy, security=security)
                figures.append(
                    {
                        "scope": "forecast",
                        "fiscal_period": label,
                        "override": ov,
                        "period_end_date": ped,
                        "fiscal_year": fiscal_year,
                        "mnemonic": mnemonic,
                        "value": val,
                        "unit": _resolved_unit(mnemonic, security),
                        "sign": _resolved_sign(mnemonic, security, val),
                        "basis": {"mnemonic": mnemonic, "definition": FIELD_BASIS.get(mnemonic, "Bloomberg DAPI field.")},
                        "vintage": vintage,
                        "source": "BBG",
                        "estimate_basis": "consensus",
                        "currency": f_cur,
                        "currency_basis": f_cur_basis,
                    }
                )
            else:
                gaps.append(
                    {
                        "fiscal_period": label,
                        "mnemonic": mnemonic,
                        "period_end_date": ped,
                        "fiscal_year": fiscal_year,
                        "override": ov,
                        "fallback_status": None,
                    }
                )

    if gaps and not sa_fallback:
        reason = "HELD_OFF: SA fallback held off by --no-sa-fallback; gap left unfilled"
        for gap in gaps:
            gap["fallback_status"] = reason
        result = {
            "record": {
                "sa_symbol_candidates": [],
                "sa_symbol": None,
                "entity_test": None,
                "classification": "HELD_OFF",
                "gaps_before": len(gaps),
                "filled_by_sa": 0,
                "sa_raw_key": None,
                "reason": reason,
                "sa_field_map": {},
            },
            "fills": [],
        }
    elif gaps:
        cw_symbol = _crosswalk_sa_symbol(s3, security)
        try:
            config = fb.load_config()
            session, _captured = fb.build_session(config)
        except Exception as e:  # noqa: BLE001 -- a session-build failure is a STOP, never a crash
            for gap in gaps:
                gap["fallback_status"] = f"STOPPED: SESSION_ERROR: {e}"
            result = {
                "record": {
                    "sa_symbol_candidates": [],
                    "sa_symbol": None,
                    "entity_test": None,
                    "classification": "SESSION_ERROR",
                    "gaps_before": len(gaps),
                    "filled_by_sa": 0,
                    "sa_raw_key": None,
                    "reason": f"STOPPED: SESSION_ERROR: {e}",
                    "sa_field_map": {},
                },
                "fills": [],
            }
        else:
            result = fb.run_fallback(
                security, sec_key, header_vals.get("NAME"), gaps, cw_symbol, s3, session, config
            )
    else:
        result = {"record": _no_gap_record(), "fills": []}

    for fill in result["fills"]:
        mnemonic = fill["mnemonic"]
        fill_cur, fill_cur_basis = _figure_currency(mnemonic, quote_ccy, reporting_ccy, security=security)
        figures.append(
            {
                "scope": "forecast",
                "fiscal_period": fill["fiscal_period"],
                "override": fill.get("override"),
                "period_end_date": fill.get("period_end_date"),
                "fiscal_year": fill.get("fiscal_year"),
                "mnemonic": mnemonic,
                "value": fill["value"],
                "unit": _resolved_unit(mnemonic, security),
                "sign": _resolved_sign(mnemonic, security, fill["value"]),
                "basis": {"mnemonic": mnemonic, "definition": FIELD_BASIS.get(mnemonic, "Bloomberg DAPI field.")},
                "vintage": vintage,
                "source": "SA_FALLBACK",
                "estimate_basis": "consensus",
                "sa_field": fill["sa_field"],
                "sa_symbol": fill["sa_symbol"],
                "sa_raw_key": fill["sa_raw_key"],
                "currency": fill_cur,
                "currency_basis": fill_cur_basis,
            }
        )

    for gap in gaps:
        if gap.get("fallback_status"):
            unresolved.append(
                {
                    "scope": "forecast",
                    "fiscal_period": gap["fiscal_period"],
                    "mnemonic": gap["mnemonic"],
                    "reason": gap["fallback_status"],
                    "fallback_status": gap["fallback_status"],
                }
            )

    return figures, unresolved, result["record"]

def _is_december_fye(hist_rows: list[dict]) -> bool:
    """AMGT-rk-109: True when EVERY row carrying a `fye_join.FUNDAMENTAL_FIELDS` member is dated
    in month 12, or when no such row exists (vacuously December -- nothing to contradict it).
    A row that carries no fundamental field (a price-only trading-day snapshot) never counts
    against this -- only a fundamentals-bearing row is the period-end record fye_join anchors
    on."""
    fye_mod = _load_fye_join()
    for row in hist_rows:
        if not any(f in row for f in fye_mod.FUNDAMENTAL_FIELDS):
            continue
        date = row.get("date")
        if not date or date[5:7] != "12":
            return False
    return True


def stage_derive(s3, security: str, raw_stamp: str | None, sa_fallback: bool = True) -> dict:
    """Read the raw bytes BACK from S3 (never the pull's in-memory result) and compute the
    derived model_block.json, landed at its own, later stamp -- independently re-runnable."""
    sec_key = security_key(security)
    if raw_stamp is None:
        raw_stamp = _select_stamp_carrying_inputs(s3, security, sec_key)
    ref = _read_json_from_s3(s3, _bronze_key("refdata", sec_key, raw_stamp))
    hist = _read_json_from_s3(s3, _bronze_key("histdata", sec_key, raw_stamp))

    header_vals = ref.get(security, {})
    header_errs = header_vals.get("__field_errors__", {})
    hist_rows = hist.get(security, [])

    # AMGT-rk-112, C-CURRENCY-LABEL: the two header currency scalars this cell reads once, up
    # front, and threads to every figure built below (header, fiscal-year and forecast alike).
    quote_ccy, reporting_ccy = _header_currencies(header_vals)

    # AMGT-rk-109: a 31 March (or any non-December) fiscal year end cannot be joined by
    # CALENDAR year (merge_by_fiscal_year mispairs the fundamentals with a price ~9 months
    # off, row W1-S2) -- fiscal_period_join anchors on the fundamentals row's own reported
    # date instead. The December path is UNCHANGED: same call, same output, byte-identical.
    is_december = _is_december_fye(hist_rows)
    non_anchor_only_fields: set[str] = set()
    fiscal_year_join_info: dict | None = None
    if is_december:
        fy_groups = merge_by_fiscal_year(hist_rows)
    else:
        fye_mod = _load_fye_join()
        fy_groups = fye_mod.fiscal_period_join(hist_rows)
        # (a) each group's PX_LAST is the paired price (never the calendar-year mispairing).
        for row in fy_groups.values():
            row["PX_LAST"] = row.get("price")
        # (b) a HIST_FIELDS member that exists ONLY on a non-anchor (price-only) row is never
        # merged by fiscal_period_join (it would be mis-dated) -- recorded under `unresolved`
        # below instead of silently dropped.
        for row in hist_rows:
            if any(f in row for f in fye_mod.FUNDAMENTAL_FIELDS):
                continue
            for k in row:
                if k != "date":
                    non_anchor_only_fields.add(k)
        # (c) the block gains a top-level fiscal_year_join key -- non-December path ONLY, so
        # the December block's bytes never change (the key's mere presence would break
        # byte-identity even with the same value, so it is never written on that path).
        fiscal_year_join_info = {
            "method": "fiscal-period (fye_join)",
            "max_gap_days": fye_mod.max_gap_days(fy_groups),
        }

    _ensure_unit_map(s3, security, sec_key, header_vals, hist_rows, raw_stamp)

    closed_fys = sorted(
        fy for fy, row in fy_groups.items()
        if any(row.get(f) not in (None, "") for f in _STATEMENT_FIELDS)
    )
    excluded_fys = sorted(fy for fy in fy_groups if fy not in closed_fys)

    vintage = _dt.datetime.utcnow().strftime("%Y-%m-%dT%H:%M:%SZ")
    figures: list[dict] = []
    unresolved: list[dict] = []

    for mnemonic in HEADER_FIELDS:
        val = header_vals.get(mnemonic)
        # AMGT-rk-37, item 2: CIE_DES_BULK's raw value is a LIST of row dicts (measured on
        # CAT) -- join it to a single string HERE, at the derive; the raw landing above is
        # untouched (it was already put to S3 in stage_pull before this function ever runs).
        if mnemonic in _BULK_TEXT_HEADER_FIELDS:
            val = _bulk_text_value(val)
        ern_left_out = None
        if mnemonic in _BULK_EARNINGS_HEADER_FIELDS:
            # CRPS-aj-1: the bulk earnings field -> [{announcement_date, period}...]; the raw landing is untouched.
            val, ern_left_out = _ern_announcements_value(val)
        if val is None or val == "":
            reason = header_errs.get(mnemonic, "empty/no value returned")
            if ern_left_out and mnemonic not in header_errs:
                reason = "no row carried a readable announcement date (%d row(s) left out)" % ern_left_out
            unresolved.append(
                {
                    "scope": "header",
                    "mnemonic": mnemonic,
                    "reason": reason,
                }
            )
            continue
        cur, cur_basis = _figure_currency(mnemonic, quote_ccy, reporting_ccy, security=security)
        basis = {"mnemonic": mnemonic, "definition": FIELD_BASIS.get(mnemonic, "Bloomberg DAPI field.")}
        if ern_left_out is not None:
            basis["rows_left_out"] = ern_left_out  # rows whose date could not be read (left out of the value)
        figures.append(
            {
                "scope": "header",
                "fiscal_year": None,
                "mnemonic": mnemonic,
                "value": val,
                "unit": _resolved_unit(mnemonic, security),
                "sign": _resolved_sign(mnemonic, security, val),
                "basis": basis,
                "vintage": vintage,
                "source": "bloomberg-dapi",
                "currency_caveat": _currency_caveat(mnemonic, security),
                "currency": cur,
                "currency_basis": cur_basis,
            }
        )

    for fy in closed_fys:
        row = fy_groups[fy]
        for mnemonic in HIST_FIELDS:
            if mnemonic not in row:
                # AMGT-rk-109, item 2(b): on the non-December path, a member that lands ONLY on
                # a non-anchor (price-only) row was never merged by fiscal_period_join (it
                # would be mis-dated) -- name it here rather than silently dropping it. The
                # December path is byte-identical: this branch is never taken there.
                if not is_december and mnemonic in non_anchor_only_fields:
                    unresolved.append(
                        {
                            "scope": "fiscal_year_line_item",
                            "fiscal_year": fy,
                            "mnemonic": mnemonic,
                            "reason": "not dated at the fiscal period end (non-December FYE, fye_join)",
                        }
                    )
                continue
            val = row[mnemonic]
            if val is None or val == "":
                unresolved.append(
                    {"scope": "fiscal_year_line_item", "fiscal_year": fy, "mnemonic": mnemonic, "reason": "empty/no value returned"}
                )
                continue
            fy_cur, fy_cur_basis = _figure_currency(mnemonic, quote_ccy, reporting_ccy, security=security)
            fig = {
                "scope": "fiscal_year_line_item",
                "fiscal_year": fy,
                "mnemonic": mnemonic,
                "value": val,
                "unit": _resolved_unit(mnemonic, security),
                "sign": _resolved_sign(mnemonic, security, val),
                "basis": {"mnemonic": mnemonic, "definition": FIELD_BASIS.get(mnemonic, "Bloomberg DAPI field.")},
                "vintage": vintage,
                "source": "bloomberg-dapi",
                "currency_caveat": _currency_caveat(mnemonic, security),
                "currency": fy_cur,
                "currency_basis": fy_cur_basis,
            }
            if not is_december and mnemonic == "PX_LAST":
                # AMGT-rk-109, item 2(a): the paired price's own observation date and elapsed
                # gap, recorded ON the figure fiscal_period_join built it from.
                fig["price_date"] = row.get("price_date")
                fig["gap_days"] = row.get("gap_days")
            figures.append(fig)

    # AMGT-rk-35 item 2(ii): every EXCLUDED (not-closed) fiscal-year group is recorded under
    # unresolved -- COUNT IN EQUALS COUNT OUT on the block's own accounting: no fiscal year
    # simply vanishes, it either lands as closed figures or is named here with its reason.
    for fy in excluded_fys:
        unresolved.append(
            {"scope": "fiscal_year_line_item", "fiscal_year": fy, "mnemonic": None, "reason": _FY_NOT_CLOSED_REASON}
        )

    end_requested = _end_requested_from_pull_log(s3, sec_key, raw_stamp)
    history_window = {
        "start": HIST_START,
        "end_requested": end_requested,
        "latest_closed_fiscal_year": max(closed_fys) if closed_fys else None,
        "rule": _FY_NOT_CLOSED_REASON,
    }

    # AMGT-rk-36, C-MODEL-FORECASTS: forecast-scope figures, their unresolved cells, and the
    # forecast_fallback record (item 2, item 5). COUNT IN EQUALS COUNT OUT here too: every one
    # of the 15 possible (period, field) cells lands as a figure or a named unresolved record.
    forecast_figures, forecast_unresolved, forecast_fallback = _derive_forecast_figures(
        s3, security, sec_key, raw_stamp, header_vals, history_window["latest_closed_fiscal_year"], vintage,
        sa_fallback=sa_fallback,
    )
    figures.extend(forecast_figures)
    unresolved.extend(forecast_unresolved)

    # AMGT-rk-37, C-MODEL-WINDOW-AND-ROWS (5): the Base case, computed ONCE here, on the
    # figures this SAME derive just built (never re-read from a different stamp).
    cases = _compute_cases(figures, quote_ccy, reporting_ccy, history_window["latest_closed_fiscal_year"])

    derived = {
        "security": security,
        "security_key": sec_key,
        "raw_stamp_used": raw_stamp,
        "fiscal_years": closed_fys,
        "fiscal_year_row_count": len(closed_fys),
        "orphan_price_rows": 0,  # by construction of merge_by_fiscal_year: every raw row's
        # fields are merged into its calendar-year group, so nothing is left un-joined; see
        # check_join.py for the independent, from-S3 re-derivation this claim is graded against.
        # Meaning now holds over the CLOSED set only (AMGT-rk-35): an excluded (not-closed)
        # fiscal year is never counted here, and is named under `unresolved` instead.
        "history_window": history_window,
        "figures": figures,
        "unresolved": unresolved,
        "forecast_fallback": forecast_fallback,
        "cases": cases,
        # AMGT-rk-112, C-CURRENCY-LABEL, RULED item 3 -- always present, both December and
        # non-December: NOTHING is converted (`converted` is always False in this trial cell).
        "currencies": {
            "quote": quote_ccy,
            "reporting": reporting_ccy,
            "mixed": quote_ccy != reporting_ccy,
            "converted": False,
        },
    }
    if fiscal_year_join_info is not None:
        # AMGT-rk-109, item 2(c): non-December path ONLY -- the December block never gains
        # this key, so its bytes are unchanged (see the comment where fiscal_year_join_info
        # is built above).
        derived["fiscal_year_join"] = fiscal_year_join_info

    derived_stamp = utc_stamp()
    derived_key = f"derived/{ENGINE}/{crosswalk.slash_safe_key(sec_key)}/{derived_stamp}/model_block.json"
    s3_put_if_absent(s3, derived_key, json.dumps(derived, default=str, indent=2).encode("utf-8"))
    return {
        "security": security,
        "sec_key": sec_key,
        "derived_stamp": derived_stamp,
        "derived_key": derived_key,
        "model_block": derived,
    }


def cmd_pull(securities: list[str], dry_run: bool = False) -> int:
    if dry_run:
        end_date = _rolling_end_date()
        best_fields = FORECAST_FIELDS + [FORECAST_DATE_FIELD]
        for sec in securities:
            sec_key = security_key(sec)
            print(
                f"DRY-RUN PULL: {sec} -> would send ReferenceDataRequest(security={sec!r}, "
                f"fields={HEADER_FIELDS})"
            )
            # CRPS-f-2: one line per historical chunk -- HIST_FIELDS split at
            # HIST_FIELDS_PER_REQUEST, order preserved.
            hist_field_chunks = _hist_field_chunks(HIST_FIELDS, HIST_FIELDS_PER_REQUEST)
            n_chunks = len(hist_field_chunks)
            for chunk_i, chunk_fields in enumerate(hist_field_chunks, start=1):
                print(
                    f"DRY-RUN PULL: {sec} -> would send HistoricalDataRequest chunk {chunk_i} "
                    f"of {n_chunks} fields={len(chunk_fields)} (security={sec!r}, "
                    f"request_fields={chunk_fields}, start_date={HIST_START}, "
                    f"end_date={end_date}, periodicity={PERIODICITY})"
                )
            for fp in FORECAST_OVERRIDES:
                print(
                    f"DRY-RUN PULL: {sec} -> would send ReferenceDataRequest(security={sec!r}, "
                    f"fields={best_fields}, overrides={{{FORECAST_OVERRIDE_FIELD!r}: {fp!r}}})"
                )
            print(

def _cases_report_figures(s3, security: str, sec_key: str, raw_stamp: str) -> tuple[list[dict], str | None, str | None]:
    """CRPS-g-1 -- the read-only reconstruction `cases-report` needs: the THREE figure kinds
    `_compute_cases` reads (the FY1 BEST_EPS forecast figure, every CLOSED fiscal year's own
    PE_RATIO figure, and the header PX_LAST figure), plus the block's own quote/reporting
    currency codes -- rebuilt from the raw bytes already landed at `raw_stamp`, the SAME
    fiscal-year join and closed-year rule `stage_derive` uses. Writes NOTHING: no
    `_ensure_unit_map` call (that is the only write inside `stage_derive`'s figure-building
    path) and no S3 put anywhere.

    FY1 BEST_EPS is read from the landed `bestdata.json` BBG value only -- this function never
    calls `_derive_forecast_figures`/the Seeking Alpha fallback session, because that is a live
    vendor call and this route's HARD BOUND is no vendor call, ever. A security whose real
    `derive` would have filled the gap through SA_FALLBACK reads ABSENT here instead (`base`:
    "no consensus EPS on the landed block") rather than making that call -- a deliberate
    divergence from `stage_derive`, never a bug."""
    ref = _read_json_from_s3(s3, _bronze_key("refdata", sec_key, raw_stamp))
    hist = _read_json_from_s3(s3, _bronze_key("histdata", sec_key, raw_stamp))
    header_vals = ref.get(security, {})
    hist_rows = hist.get(security, [])
    quote_ccy, reporting_ccy = _header_currencies(header_vals)

    is_december = _is_december_fye(hist_rows)
    if is_december:
        fy_groups = merge_by_fiscal_year(hist_rows)
    else:
        fye_mod = _load_fye_join()
        fy_groups = fye_mod.fiscal_period_join(hist_rows)
        for row in fy_groups.values():
            row["PX_LAST"] = row.get("price")

    closed_fys = sorted(
        fy for fy, row in fy_groups.items()
        if any(row.get(f) not in (None, "") for f in _STATEMENT_FIELDS)
    )

    figures: list[dict] = []

    px_val = header_vals.get("PX_LAST")
    if px_val not in (None, ""):
        figures.append({"scope": "header", "mnemonic": "PX_LAST", "value": px_val, "vintage": None})

    # AMGT-uu-3: the Base case's adjusted P/E reads each closed year's PX_LAST and
    # IS_COMP_EPS_ADJUSTED (joined on fiscal_year), the same two figures `stage_derive` lands.
    for fy in closed_fys:
        for mnemonic in ("PX_LAST", "IS_COMP_EPS_ADJUSTED"):
            val = fy_groups[fy].get(mnemonic)
            if isinstance(val, (int, float)):
                figures.append(
                    {"scope": "fiscal_year_line_item", "fiscal_year": fy, "mnemonic": mnemonic, "value": val}
                )

    try:
        best = _read_json_from_s3(s3, _bronze_key("bestdata", sec_key, raw_stamp))
    except botocore.exceptions.ClientError:
        best = None
    if best:
        fy1_row = (best.get("1FY") or {}).get(security) or {}
        eps_val = fy1_row.get("BEST_EPS")
        if eps_val not in (None, ""):
            ped = fy1_row.get(FORECAST_DATE_FIELD)
            figures.append(
                {
                    "scope": "forecast",
                    "mnemonic": "BEST_EPS",
                    "fiscal_period": "FY1",
                    "fiscal_year": _fiscal_year_from_period_end(ped) if ped else None,
                    "period_end_date": ped,
                    "value": eps_val,
                    "source": "BBG",
                    "basis": {"mnemonic": "BEST_EPS", "definition": FIELD_BASIS.get("BEST_EPS", "Bloomberg DAPI field.")},
                }
            )

    return figures, quote_ccy, reporting_ccy


def _stamp_carries_inputs(s3, security: str, sec_key: str, stamp: str) -> bool:
    """CRPS-j-3: True when the landed raw at `stamp` carries the INPUTS of the Base-case rule
    (`lib.research.api.case_rule.compute_cases`) -- i.e. the rule, run over that stamp's figures,
    does NOT refuse for missing consensus EPS or missing adjusted P/E history. Those two input
    refusals are exactly the ABSENT results that carry no `cause`; every other result (SERVED, or
    a refusal with a `cause`: forward EPS not positive, no price, currency mismatch, outside band)
    was computed FROM the inputs, so a currency refusal is a stamp that carries them. Reads landed
    raw only, through `_cases_report_figures` (no write, no vendor call)."""
    figures, quote_ccy, reporting_ccy = _cases_report_figures(s3, security, sec_key, stamp)
    base = _compute_cases(figures, quote_ccy, reporting_ccy)["base"]
    return base.get("status") == "SERVED" or bool(base.get("cause"))


def _select_stamp_carrying_inputs(s3, security: str, sec_key: str) -> str:
    """CRPS-j-3: the raw stamp `derive` and `cases-report` read -- the newest qualifying stamp
    (`_stamps_with_raw_pair`, newest first) that carries the rule's inputs. When no landed stamp
    carries them, the newest qualifying stamp exactly as `_latest_stamp` returns it, so the
    refusal and its reason are unchanged; when the newest already carries them, that same newest
    stamp (no older stamp is read). Sits ON TOP of `_latest_stamp`'s pair rule and its errors."""
    newest = _latest_stamp(s3, sec_key)
    if _stamp_carries_inputs(s3, security, sec_key, newest):
        return newest
    for stamp in _stamps_with_raw_pair(s3, sec_key):
        if stamp == newest:
            continue
        try:
            if _stamp_carries_inputs(s3, security, sec_key, stamp):
                return stamp
        except botocore.exceptions.ClientError:
            continue  # an unreadable older stamp does not carry; fall through to the next / newest
    return newest


def cmd_stamp_census(securities: list[str]) -> int:
    """CRPS-j-3 -- `stamp-census`: READ-ONLY. For each security, over every stamp
    `_stamps_with_raw_pair` returns (newest first), report whether the newest carries the rule's
    inputs and which stamp `_select_stamp_carrying_inputs` would read. The three buckets
    (newest_carries / older_carries / none_carries) partition the keys read. Writes NOTHING."""
    s3 = _s3()
    newest_carries = older_carries = none_carries = 0
    for sec in securities:
        sec_key = security_key(sec)
        newest = _latest_stamp(s3, sec_key)
        newest_yes = _stamp_carries_inputs(s3, sec, sec_key, newest)
        carrying = newest if newest_yes else _select_stamp_carrying_inputs(s3, sec, sec_key)
        if newest_yes:
            newest_carries += 1
        elif carrying != newest:
            older_carries += 1
        else:
            carrying = None
            none_carries += 1
        print(
            f"stamp {sec_key} newest {newest} newest_carries {'yes' if newest_yes else 'no'} "
            f"carrying {carrying if carrying else 'none'}"
        )
    print(
        f"stamp_census keys {len(securities)} newest_carries {newest_carries} "
        f"older_carries {older_carries} none_carries {none_carries} "
        f"sum {newest_carries + older_carries + none_carries}"
    )
    return 0


def _cases_report_unresolved(s3, securities: list[str]) -> list[tuple[str, str]]:
    """CRPS-l-4: (input, bronze prefix) for every security whose landed raw `cases-report`
    cannot resolve. The landed raw is keyed by the Bloomberg form ("AMD US Equity"); the
    underscore form maps to the SAME sec_key and so finds stamps, but its own string is no key
    inside the landed refdata/histdata, so it read as an empty block and a false ABSENT. Read-only."""
    bad: list[tuple[str, str]] = []
    for sec in securities:
        sec_key = security_key(sec)
        prefix = _bronze_prefix("refdata", sec_key)
        try:
            stamp = _latest_stamp(s3, sec_key)
            figures, _q, _r = _cases_report_figures(s3, sec, sec_key, stamp)
        except (RuntimeError, KeyError, botocore.exceptions.ClientError):
            bad.append((sec, prefix))
            continue
        if not figures:
            bad.append((sec, prefix))
    return bad


def cmd_cases_report(securities: list[str]) -> int:
    """CRPS-g-1 -- `cases-report`: for each security, read the NEWEST landed raw stamp from
    S3, rebuild the figures and cases in memory exactly as `derive` would (modulo the
    no-vendor-call divergence `_cases_report_figures` documents), and print one `cases` line
    plus the summary. Writes NOTHING -- the only S3 touches are GetObject/ListObjectsV2 reads
    of already-landed raw."""
    s3 = _s3()
    unresolved = _cases_report_unresolved(s3, securities)
    if unresolved:
        for sec, prefix in unresolved:
            print(
                f'REFUSE: {sec} resolves to no landed raw stamp under {prefix} '
                f'(pass the Bloomberg form, e.g. "AMD US Equity")'
            )
        return 2
    served = 0
    refused = 0
    for sec in securities:
        sec_key = security_key(sec)
        raw_stamp = _select_stamp_carrying_inputs(s3, sec, sec_key)
        figures, quote_ccy, reporting_ccy = _cases_report_figures(s3, sec, sec_key, raw_stamp)
        cases = _compute_cases(figures, quote_ccy, reporting_ccy)
        base = cases["base"]
        status = base.get("status")
        ratio = base.get("ratio_to_px_last")
        cause = base.get("cause")
        reason = base.get("reason")
        line = (
            f"cases {sec_key} base {status} ratio {ratio if ratio is not None else 'none'} "
            f"cause {cause if cause else 'none'} reason {reason}"
        )
        band_basis = base.get("band_basis")
        if band_basis:
            low = band_basis["low"]
            high = band_basis["high"]
            line += (
                f" band_basis low {low['value']} {low['calibration']} "
                f"high {high['value']} {high['calibration']}"
            )
        print(line)
        if status == "SERVED":
            served += 1
        else:
            refused += 1
    print(f"cases_report served {served} refused {refused}")
    return 0

def _registered_security_keys() -> list[str]:
    """CRPS-t-1: every security the live research library resolves through the served
    crosswalk -- `backlog.build(library_root())` records carrying a `security_key`
    (<ROOT>_<CC>_Equity, deduplicated, sorted)."""
    from lib.research.corpus import backlog as _backlog
    from lib.research.corpus import library as _library

    document = _backlog.build(_library.library_root())
    return sorted({r["security_key"] for r in document["records"] if r.get("security_key")})


def _model_block_keys_in_store(s3) -> set[str]:
    """The security_key (IDENTITY form, '/' kept) of every object under derived/model_block/
    in the producer's own bucket. C-SLASH-SAFE-KEY: stage_derive now writes the ENCODED
    segment (`crosswalk.slash_safe_key`), so a store key maps back to its identity with
    the crosswalk's exact inverse `security_key_from_slash_safe`; a LEGACY raw-slash
    partition (live until the round-25 move: `BP/_LN_Equity/<stamp>/<file>`) is already
    the identity, so it is kept as found -- both forms of one name yield ONE identity.
    The segment is everything between the prefix and `<stamp>/<file>`.
    One paginated list, no delimiter (a '/' inside a legacy key would truncate a delimited list)."""
    # slash-safe: not a per-security path (a PREFIX listing; it DECODES each stored segment with security_key_from_slash_safe and builds no per-security path)
    prefix = f"derived/{ENGINE}/"
    found: set[str] = set()
    token = None
    while True:
        kwargs = {"Bucket": BUCKET, "Prefix": prefix}
        if token:
            kwargs["ContinuationToken"] = token
        resp = s3.list_objects_v2(**kwargs)
        for obj in resp.get("Contents", []):
            rest = obj["Key"][len(prefix):]
            if rest.count("/") >= 2:
                segment = rest.rsplit("/", 2)[0]
                if "/" not in segment:
                    try:
                        segment = crosswalk.security_key_from_slash_safe(segment)
                    except crosswalk.SlashSafeKeyError:
                        pass  # not a slash-safe form (outer whitespace / non-ASCII): keep the segment as found
                found.add(segment)
        if resp.get("IsTruncated"):
            token = resp.get("NextContinuationToken")
        else:
            return found


def _returned_records(name: str, result) -> int:
    """AMENDMENT 2 (CRPS-ap-7, round 40): the RETURNED RECORDS one response carries, the budget ledger's own unit.
    A reference request returns {security: {...}}: one record per security row. A history request returns
    {security: [dated rows]}: one record per dated row. Counted from the response received, never estimated;
    any other shape, or a response that cannot be read, counts 0 (bookkeeping never fails a request)."""
    try:
        if not isinstance(result, dict):
            return 0
        if name == "reference_data":
            return len(result)
        if name == "historical_data":
            return sum(len(rows) for rows in result.values() if isinstance(rows, list))
    except Exception:  # noqa: BLE001 -- counting must never break a request
        return 0
    return 0


class _CountingClient:
    """Delegates to the estate client and counts each Bloomberg request it sends, and (AMENDMENT 2, CRPS-ap-7,
    round 40) the returned records of each response."""

    def __init__(self, inner) -> None:
        self._inner = inner
        self.calls = 0
        self.records = 0

    def __getattr__(self, name):
        attr = getattr(self._inner, name)
        if name in ("reference_data", "historical_data", "intraday_bars") and callable(attr):
            def _counted(*a, **kw):
                self.calls += 1
                result = attr(*a, **kw)
                self.records += _returned_records(name, result)
                return result
            return _counted
        return attr


BUDGET_MODE = "amgt-model-block"
# The estate's own rule for finding the ledger module (integrations/bloomberg/pipeline/etf_daily_pull.py): a glob under the
# repository's data/ tree, exactly one hit -- no hand-kept firm name here.
BUDGET_LEDGER_GLOB = ("data", "*", "scripts", "rivalfunds", "bbg_budget.py")


def record_fire_budget(*, calls, security_fields, figures, records, stamp) -> bool:
    """AMENDMENT 2 (CRPS-ap-7, round 40; CTO rulings msg 31290, 31295, 31310): append ONE line for this fire to the shared
    Bloomberg budget ledger (`bbg_budget.py` under `data/*/scripts/rivalfunds/`, loaded BY PATH from the repository this file
    sits in, the way `_load_build_unit_map` loads its sibling). AMENDMENT 3 (CRPS-ap-7, round 40; CTO ruling msg 31815):
    `rows` = `records` -- the ledger holds ONE unit, RETURNED RECORDS, as its other writers record rows returned; a
    figures-sized line (about 67,000) would overstate the day's spend about 16-fold and could refuse another product
    capacity. The batch_id carries all four counts and ends `rows_unit=returned-records`. Returns what `bbg_budget.record` returns. NEVER raises: a missing module, an unreadable path or a failed
    write returns False -- a fire is never blocked or failed by bookkeeping."""
    try:
        repo = Path(__file__).resolve().parent.parent.parent.parent
        hits = sorted(repo.glob("/".join(BUDGET_LEDGER_GLOB)))
        if len(hits) != 1:
            return False
        mod_path = hits[0]
        spec = importlib.util.spec_from_file_location("bbg_budget_ledger", mod_path)
        if spec is None or spec.loader is None:
            return False
        mod = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(mod)
        batch_id = (
            f"{BUDGET_MODE}:{stamp}:calls={int(calls)}:security_fields={int(security_fields)}"
            f":figures={int(figures)}:records={int(records)}:rows_unit=returned-records"
        )
        return bool(mod.record(BUDGET_MODE, int(records), batch_id))
    except Exception:  # noqa: BLE001 -- bookkeeping never fails a fire
        return False


def _figure_count(derived: dict) -> int:
    """The figures in one derived block (stage_derive's return); 0 when the block carries none."""
    try:
        return len(derived["model_block"]["figures"])
    except Exception:  # noqa: BLE001
        return 0


def _close_fire_budget(counter: "_CountingClient", pulled: int, figures: int) -> None:
    """AMENDMENT 2 (CRPS-ap-7, round 40): the one budget record at the end of a real fire, and its one printed line.
    Never raises and never changes the fire's rc."""
    try:
        per_name = len(HEADER_FIELDS) + len(HIST_FIELDS) + len(FORECAST_FIELDS) * len(FORECAST_OVERRIDES)
        security_fields = pulled * per_name
        written = record_fire_budget(
            calls=counter.calls, security_fields=security_fields, figures=figures,
            records=counter.records, stamp=utc_stamp(),
        )
        print(
            f"budget_record {BUDGET_MODE} rows {counter.records} records {counter.records} calls {counter.calls} "
            f"security_fields {security_fields} written {1 if written else 0}",
            flush=True,
        )
    except Exception:  # noqa: BLE001
        pass


def cmd_admit(dry_run: bool = False) -> int:
    """CRPS-t-1: pull + derive every registered name that has no model block yet."""
    registered = _registered_security_keys()
    s3 = _s3()
    present = _model_block_keys_in_store(s3)
    missing = [k for k in registered if k not in present]
    n, k_missing = len(registered), len(missing)
    print(f"admit population registered {n} without_block {k_missing}", flush=True)
    if dry_run:
        for key in missing:
            print(f"DRY-RUN ADMIT: {key} -> would pull then derive")
        print(f"admit population registered {n} without_block {k_missing} pulled 0 bloomberg_calls 0")
        return 0
    if k_missing == 0:
        print(f"admit population registered {n} without_block 0 pulled 0 bloomberg_calls 0")
        return 0
    try:
        blp = _load_blp_client()
        status = blp.health()
        reachable = bool(status.get("ok"))
    except Exception:  # noqa: BLE001 -- any failure to open a session is the same refusal
        reachable = False
    if not reachable:
        print(ADMIT_REFUSED_LINE)
        return 4
    counter = _CountingClient(blp)
    pulled = 0
    failed = 0
    figures_total = 0  # AMENDMENT 2 (CRPS-ap-7, round 40): figures in the blocks this run derived
    for i in range(0, k_missing, ADMIT_BATCH):
        batch = missing[i:i + ADMIT_BATCH]
        for key in batch:
            security = key.replace("_", " ")
            try:
                r = stage_pull(counter, s3, security)
                pulled += 1
                d = stage_derive(s3, security, raw_stamp=r["raw_stamp"])
                figures_total += _figure_count(d)
                print(f"ADMIT: {security} -> raw {r['raw_stamp']} derived {d['derived_stamp']}")
            except Exception as exc:  # noqa: BLE001 -- one name failing must not hide the others
                failed += 1
                print(f"ADMIT FAILED: {security}: {type(exc).__name__}: {exc}")
    # AMENDMENT 2 (CRPS-ap-7, round 40): call site 1 of 2 -- ONE ledger line per real admit that pulled a name. Its printed
    # line comes BEFORE the summary line so the summary stays the last line of output (round 34's tests read lines[-1]).
    if pulled >= 1:
        _close_fire_budget(counter, pulled, figures_total)
    print(
        f"admit population registered {n} without_block {k_missing} "
        f"pulled {pulled} bloomberg_calls {counter.calls}"
    )
    return 1 if failed else 0


def _model_block_latest_stamps(s3) -> dict[str, str]:
    """CRPS-aj-1: security_key (IDENTITY form) -> the newest `<stamp>` under derived/model_block/<key>/.
    The same one paginated prefix listing and the same segment decoding as `_model_block_keys_in_store`."""
    # slash-safe: not a per-security path (a PREFIX listing; it DECODES each stored segment with security_key_from_slash_safe and builds no per-security path)
    prefix = f"derived/{ENGINE}/"
    latest: dict[str, str] = {}
    token = None
    while True:
        kwargs = {"Bucket": BUCKET, "Prefix": prefix}
        if token:
            kwargs["ContinuationToken"] = token
        resp = s3.list_objects_v2(**kwargs)
        for obj in resp.get("Contents", []):
            rest = obj["Key"][len(prefix):]
            if rest.count("/") >= 2:
                segment, stamp, _file = rest.rsplit("/", 2)
                if "/" not in segment:
                    try:
                        segment = crosswalk.security_key_from_slash_safe(segment)
                    except crosswalk.SlashSafeKeyError:
                        pass
                if stamp > latest.get(segment, ""):
                    latest[segment] = stamp
        if resp.get("IsTruncated"):
            token = resp.get("NextContinuationToken")
        else:
            return latest


def _pull_log_latest_stamps(s3, keys: list[str]) -> dict[str, str]:
    """CRPS-ao-1: security_key -> the newest PULL stamp (the pull-log state object a pull lands, same stamp as its
    bronze objects). ONE paginated listing of the pull_log table prefix for the whole population; each object's
    folder is matched back to a key through `_state_pull_log_prefix`, so no segment is decoded."""
    by_prefix = {_state_pull_log_prefix(k): k for k in keys}
    if not by_prefix:
        return {}
    table = hstore.table_prefix("amgt", "state", "pull_log")
    latest: dict[str, str] = {}
    token = None
    while True:
        kwargs = {"Bucket": BUCKET, "Prefix": table}
        if token:
            kwargs["ContinuationToken"] = token
        resp = s3.list_objects_v2(**kwargs)
        for obj in resp.get("Contents", []) or []:
            folder, _, name = obj["Key"].rpartition("/")
            key = by_prefix.get(folder + "/")
            if key is not None and name.endswith(".json"):
                stamp = name[:-len(".json")]
                if stamp > latest.get(key, ""):
                    latest[key] = stamp
        if resp.get("IsTruncated"):
            token = resp.get("NextContinuationToken")
        else:
            return latest


def cmd_refresh(dry_run: bool = False) -> int:
    """CRPS-aj-1: re-pull then re-derive every registered name that HAS a model block, except a name whose
    last PULL stamp is already today's (UTC) date -- so the same run's `admit` and a same-day re-fire
    never pull twice. A same-day re-derive does not count as a pull. Same batching, health check and rc contract as `cmd_admit`."""
    registered = _registered_security_keys()
    s3 = _s3()
    present = _model_block_keys_in_store(s3)
    with_block = [k for k in registered if k in present]
    n, k_with = len(registered), len(with_block)
    print(f"refresh population registered {n} with_block {k_with}", flush=True)
    print(f"refresh fields header {len(HEADER_FIELDS)} hist {len(HIST_FIELDS)} forecast {len(FORECAST_FIELDS)}", flush=True)
    today = _dt.datetime.utcnow().strftime("%Y%m%d")
    # AMENDMENT (CRPS-ao-1, round 39): the skip keys on the last PULL stamp, never the derived stamp -- a same-day
    # re-derive writes a today-dated derived stamp and made the scheduled fire skip every name (row Z-S88).
    latest = _pull_log_latest_stamps(s3, with_block)
    todo: list[str] = []
    current = 0
    for key in with_block:
        if latest.get(key, "")[:8] == today:  # pulled today
            current += 1
            print(f"REFRESH SKIP: {key} already_current")
        else:
            todo.append(key)
    if dry_run:
        for key in todo:
            print(f"DRY-RUN REFRESH: {key} -> would pull then derive")
        print(f"refresh population registered {n} with_block {k_with} already_current {current} pulled 0 bloomberg_calls 0")
        return 0
    if not todo:
        print(f"refresh population registered {n} with_block {k_with} already_current {current} pulled 0 bloomberg_calls 0")
        return 0
    try:
        blp = _load_blp_client()
        status = blp.health()
        reachable = bool(status.get("ok"))
    except Exception:  # noqa: BLE001 -- any failure to open a session is the same refusal
        reachable = False
    if not reachable:
        print(ADMIT_REFUSED_LINE)
        return 4
    counter = _CountingClient(blp)
    pulled = 0
    failed = 0
    figures_total = 0  # AMENDMENT 2 (CRPS-ap-7, round 40): figures in the blocks this run derived
    for i in range(0, len(todo), ADMIT_BATCH):
        batch = todo[i:i + ADMIT_BATCH]
        for key in batch:
            security = key.replace("_", " ")
            try:
                r = stage_pull(counter, s3, security)
                pulled += 1
                # Issuer ruling, round 34 (2.0.0_SA_COO~X): the daily refresh is a BLOOMBERG pull (Andrew-direct
                # 2026-10-03 ~20:3xZ card 3) and never makes the Seeking Alpha fallback call -- an empty BEST cell
                # reads `unresolved` with its reason. Measured at the ruling: 0 SA_FALLBACK figures in 102 latest
                # served blocks, so nothing served is lost.
                d = stage_derive(s3, security, raw_stamp=r["raw_stamp"], sa_fallback=False)
                figures_total += _figure_count(d)
                print(f"REFRESH: {security} -> raw {r['raw_stamp']} derived {d['derived_stamp']}")
            except Exception as exc:  # noqa: BLE001 -- one name failing must not hide the others
                failed += 1
                print(f"REFRESH FAILED: {security}: {type(exc).__name__}: {exc}")
    # AMENDMENT 2 (CRPS-ap-7, round 40): call site 2 of 2 -- ONE ledger line per real refresh that reached Bloomberg
    # (its printed line sits before the summary line, which stays last: round 34's tests read lines[-1]).
    _close_fire_budget(counter, pulled, figures_total)
    print(
        f"refresh population registered {n} with_block {k_with} already_current {current} "
        f"pulled {pulled} bloomberg_calls {counter.calls}"
    )
    return 1 if failed else 0


def main(argv: list[str]) -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("stage", choices=["pull", "derive", "cases-report", "stamp-census", "admit", "refresh"])
    ap.add_argument("--securities", nargs="+", default=None)
    ap.add_argument("--dry-run", action="store_true")
    ap.add_argument("--no-sa-fallback", action="store_true",
                    help="derive only: never call the Seeking Alpha forecast fallback (CRPS-y-2)")
    args = ap.parse_args(argv)
    if args.stage == "admit":
        if args.securities:
            ap.error("admit takes no --securities (its population is the registered library)")
        return cmd_admit(dry_run=args.dry_run)
    if args.stage == "refresh":
        if args.securities:
            ap.error("refresh takes no --securities (its population is the registered names that have a block)")
        return cmd_refresh(dry_run=args.dry_run)
    if not args.securities:
        ap.error("the following arguments are required: --securities")
    if args.stage == "stamp-census":
        return cmd_stamp_census(args.securities)  # read-only: the F2/G-24 cap guards vendor pulls
    if len(args.securities) > 3:
        raise SystemExit("REFUSE: more than 3 securities requested (F2/G-24 cap)")
    if args.stage == "cases-report":
        return cmd_cases_report(args.securities)
    if args.stage == "pull":
        return cmd_pull(args.securities, dry_run=args.dry_run)
    return cmd_derive(args.securities, dry_run=args.dry_run, sa_fallback=not args.no_sa_fallback)


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))
